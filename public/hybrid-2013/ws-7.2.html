จุดประสงค์

เรียก API ด้วยเมธอด POST PATCH และ DELETE พร้อมจัดการผลและข้อผิดพลาด และอัปเดตหน้าจอให้สอดคล้องกับผลที่ได้

คำชี้แจง

ionic serve ต้องรันอยู่ และต้องต่ออินเทอร์เน็ต ทุกขั้นที่ "วางโค้ดทั้งไฟล์" ให้ลบของเดิมก่อนด้วย Ctrl+A แล้ว Delete ใบงานนี้ใช้ JSONPlaceholder (API จำลองที่รับคำขอแต่ไม่บันทึกข้อมูลจริง) เพื่อฝึกส่ง แก้ไข และลบข้อมูล ทำใบงาน 7.1 ให้เสร็จก่อน (ใช้ไฟล์ http.ts ร่วมกัน)

  5. สร้างหน้าจัดการโพสต์ src/pages/Posts.tsx วางโค้ดนี้ทั้งไฟล์ กด Ctrl+S

    ไฟล์ src/pages/Posts.tsx
    import { useEffect, useState } from 'react';
import {
  IonButton, IonContent, IonHeader, IonInput, IonItem, IonLabel, IonList, IonPage, IonSpinner, IonTitle, IonToolbar,
  useIonAlert,
} from '@ionic/react';
import { createPost, deletePost, listPosts, updatePost, Post } from '../api/posts';

let localId = 1000; // id ชั่วคราวฝั่งแอป เพราะ JSONPlaceholder ตอบ id 101 ซ้ำทุกครั้ง

const Posts: React.FC = () => {
  const [presentAlert] = useIonAlert();
  const [posts, setPosts] = useState<Post[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState('');
  const [title, setTitle] = useState('');
  const [busy, setBusy] = useState(false);

  useEffect(() => {
    const ctrl = new AbortController();
    listPosts(ctrl.signal)
      .then(setPosts)
      .catch(e => {
        if (e.name !== 'AbortError') setError(e instanceof Error ? e.message : 'เกิดข้อผิดพลาด');
      })
      .finally(() => {
        if (!ctrl.signal.aborted) setLoading(false);
      });
    return () => ctrl.abort();
  }, []);

  async function add() {
    if (title.trim() === '' || busy) return;
    setBusy(true);
    try {
      const created = await createPost(title.trim(), 'เนื้อหาตัวอย่าง');
      setPosts([{ ...created, id: localId++ }, ...posts]);
      setTitle('');
      setError('');
    } catch (e) {
      setError(e instanceof Error ? e.message : 'เกิดข้อผิดพลาด');
    } finally {
      setBusy(false);
    }
  }

  async function rename(p: Post) {
    setBusy(true);
    try {
      const updated = await updatePost(p.id, { title: p.title + ' (แก้แล้ว)' });
      setPosts(posts.map(x => (x.id === p.id ? { ...x, title: updated.title } : x)));
      setError('');
    } catch (e) {
      setError(e instanceof Error ? e.message : 'เกิดข้อผิดพลาด');
    } finally {
      setBusy(false);
    }
  }

  function confirmRemove(p: Post) {
    presentAlert({
      header: 'ลบโพสต์?',
      message: p.title,
      buttons: [
        'ยกเลิก',
        {
          text: 'ลบ',
          role: 'destructive',
          handler: async () => {
            try {
              await deletePost(p.id);
              setPosts(posts.filter(x => x.id !== p.id));
            } catch (e) {
              setError(e instanceof Error ? e.message : 'เกิดข้อผิดพลาด');
            }
          },
        },
      ],
    });
  }

  return (
    <IonPage>
      <IonHeader>
        <IonToolbar color="primary">
          <IonTitle>โพสต์ ({posts.length})</IonTitle>
        </IonToolbar>
      </IonHeader>
      <IonContent className="ion-padding">
        <IonInput
          label="ชื่อเรื่องใหม่"
          labelPlacement="floating"
          fill="outline"
          value={title}
          onIonInput={e => setTitle(String(e.detail.value ?? ''))}
        />
        <IonButton expand="block" onClick={add} disabled={busy || title.trim() === ''}>เพิ่มโพสต์</IonButton>

        {loading && <p className="ion-text-center"><IonSpinner /> กำลังโหลด...</p>}
        {error && <p>{error}</p>}

        <IonList>
          {posts.map(p => (
            <IonItem key={p.id}>
              <IonLabel className="ion-text-wrap">{p.title}</IonLabel>
              <IonButton size="small" fill="outline" disabled={busy} onClick={() => rename(p)}>แก้ชื่อ</IonButton>
              <IonButton size="small" color="danger" onClick={() => confirmRemove(p)}>ลบ</IonButton>
            </IonItem>
          ))}
        </IonList>
      </IonContent>
    </IonPage>
  );
};

export default Posts;
    ผลที่ควรเห็น: ยังไม่เห็นผลบนหน้าเว็บ

  7. เปิดที่อยู่ /posts ในเบราว์เซอร์

    ผลที่ควรเห็น: แถบหัว โพสต์ (10) ช่อง ชื่อเรื่องใหม่ ปุ่ม เพิ่มโพสต์ (จางกดไม่ได้เพราะช่องว่าง) และรายการโพสต์ 10 รายการ แต่ละรายการมีปุ่ม แก้ชื่อ และ ลบ

  8. ทดสอบสร้าง: พิมพ์ ทดสอบโพสต์ใหม่ ในช่อง แล้วกดปุ่ม เพิ่มโพสต์ จากนั้นเปิดแท็บ Network ของ DevTools ดูคำขอ POST

    ผลที่ควรเห็น: โพสต์ใหม่อยู่บนสุดของรายการ หัวข้อเป็น โพสต์ (11) ช่องถูกล้าง และใน Network เห็นคำขอ POST ได้รหัสสถานะ 201 ในคำตอบมี id เป็น 101

  9. สร้างโพสต์ที่สองอีกหนึ่งรายการ แล้วดูคำตอบใน Network ว่า id เป็นอะไร

    ผลที่ควรเห็น: id ในคำตอบเป็น 101 เหมือนเดิม (API จำลองตอบ id เดิมเสมอ) แต่ในรายการทั้งสองโพสต์แสดงได้ปกติ เพราะโค้ดใช้ id ชั่วคราวของแอป (localId) เป็น key

  10. ทดสอบแก้: กดปุ่ม แก้ชื่อ ที่โพสต์ใดโพสต์หนึ่ง ดูคำขอ PATCH ใน Network

    ผลที่ควรเห็น: ชื่อโพสต์นั้นมีคำว่า (แก้แล้ว) ต่อท้าย คำขอเป็น PATCH ได้รหัสสถานะ 200 (หมายเหตุ: โพสต์ที่สร้างเองมี id ชั่วคราวเกิน 1000 จึงอาจได้ 500 จากเซิร์ฟเวอร์จำลอง ให้ทดสอบกับโพสต์ 10 รายการแรกที่มาจาก API)

  11. ทดสอบลบ: กดปุ่ม ลบ ที่โพสต์หนึ่ง

    ผลที่ควรเห็น: ขึ้นกล่อง ลบโพสต์? แสดงชื่อโพสต์ พร้อมปุ่ม ยกเลิก และ ลบ กด ยกเลิก โพสต์ยังอยู่ กด ลบ ในกล่องแล้ว โพสต์หาย และใน Network เห็นคำขอ DELETE รหัสสถานะ 200

  12. รีเฟรชหน้า (F5) แล้วดูรายการ

    ผลที่ควรเห็น: รายการกลับมาเป็นโพสต์ 10 รายการเดิม โพสต์ที่เพิ่ม แก้ และลบหายไปหมด เพราะ JSONPlaceholder ไม่บันทึกข้อมูลจริง เขียนในช่องบันทึกว่า API จริงที่มีฐานข้อมูลจะต่างอย่างไร

  13. ทดสอบความผิดพลาด 1: ที่แท็บ Network ตั้งเป็น Offline แล้วกดปุ่ม เพิ่มโพสต์ (พิมพ์ชื่อไว้ก่อน)

    ผลที่ควรเห็น: ขึ้นข้อความ เชื่อมต่อเครือข่ายไม่ได้ ไม่มีโพสต์เพิ่มในรายการ และปุ่มกลับมากดได้หลังเสร็จ (ปุ่มถูกปิดขณะรอเพื่อกันการกดซ้ำ) จากนั้นตั้งกลับเป็น No throttling

  14. ทดสอบความผิดพลาด 2: ในไฟล์ posts.ts ชั่วคราวเปลี่ยนคำว่า /posts/${id} ในฟังก์ชัน deletePost เป็น /postz/${id} แล้วกดลบโพสต์

    ผลที่ควรเห็น: ขึ้นข้อความ เซิร์ฟเวอร์ตอบกลับผิดพลาด (404) และโพสต์ยังอยู่ในรายการ จากนั้นแก้กลับให้ถูกต้อง

  15. ทำตาราง 3 แถว (สร้าง แก้ไข ลบ) ในไฟล์ Word: เมธอด / URL / เนื้อหาที่ส่ง / รหัสสถานะที่ได้ (ดูจาก Network) / ข้อความที่ผู้ใช้เห็น

    ผลที่ควรเห็น: ตารางครบ 3 แถว (สร้างได้ 201 ส่วนแก้และลบได้ 200)

  16. ถ่ายภาพหน้าจอ (รายการ ฟอร์ม และกรณีออฟไลน์) แล้วส่งงานขึ้น GitHub

คำแนะนำ

  • ถ้า PATCH หรือ DELETE ถูกบล็อก ให้ตรวจว่าใช้ HTTPS และพิมพ์ URL ถูก
  • ปิดปุ่มขณะรอคำตอบเพื่อกันการกดซ้ำ แล้วเปิดปุ่มคืนใน finally
  • การแก้ state ของรายการต้องสร้างอาเรย์ใหม่ด้วย map หรือ filter
