ใบงานที่ 8.1 ออกแบบกรณีทดสอบและทดสอบโปรแกรมย่อยด้วย Vitest

เวลา 150 นาที · งานเดี่ยว · คะแนนเต็ม 10

จุดประสงค์

ออกแบบกรณีทดสอบ เขียนชุดทดสอบโปรแกรมย่อยด้วย Vitest ที่ล้มเมื่อโค้ดผิด และรายงานข้อผิดพลาดที่พบได้

คำชี้แจง

เริ่มจากทดสอบก่อนเขียนโค้ด (test-first): ใช้ไฟล์ทดสอบตั้งต้นเป็นข้อกำหนดของฟังก์ชัน validateAmount และ sumAmount เขียนฟังก์ชันให้ผ่านทุกการทดสอบ จากนั้นทำแบบเดียวกันกับฟังก์ชันตรรกะของแอปตนเอง

สิ่งที่ต้องเตรียม

  • ผลงานใบงาน 4.2 และ 6.3
  • ใบความรู้ที่ 8 หัวข้อกรณีทดสอบและ Vitest
  • โปรเจกต์ที่สร้างจาก ionic start ซึ่งมี Vitest อยู่แล้ว (ตรวจว่า npm run test:unit ทำงาน)

โค้ดตั้งต้น

คัดลอกไปวางในไฟล์ที่ระบุในโค้ด
import { describe, expect, it } from 'vitest';
import { sumAmount, validateAmount } from './validate';

describe('validateAmount', () => {
  it('ยอมรับจำนวนเงินที่ถูกต้อง', () => {
    expect(validateAmount('45')).toBe('');
    expect(validateAmount('0.01')).toBe('');
  });

  it('ปฏิเสธค่าว่างและช่องว่าง', () => {
    expect(validateAmount('')).toBe('กรุณากรอกจำนวนเงิน');
    expect(validateAmount('   ')).toBe('กรุณากรอกจำนวนเงิน');
  });

  it('ปฏิเสธข้อความที่ไม่ใช่ตัวเลข', () => {
    expect(validateAmount('abc')).toBe('จำนวนเงินต้องเป็นตัวเลข');
  });

  it('ปฏิเสธศูนย์และค่าติดลบ (ค่าขอบเขต)', () => {
    expect(validateAmount('0')).toBe('จำนวนเงินต้องมากกว่า 0');
    expect(validateAmount('-5')).toBe('จำนวนเงินต้องมากกว่า 0');
  });

  it('ปฏิเสธค่าที่เกินกำหนด (ค่าขอบเขต)', () => {
    expect(validateAmount('1000000')).toBe('');
    expect(validateAmount('1000001')).toBe('จำนวนเงินต้องไม่เกิน 1,000,000');
  });

  // TODO: เพิ่มกรณีทศนิยมเกิน 2 ตำแหน่ง เช่น '1.234'
  // TODO: เพิ่มกรณีตัวเลขที่มีช่องว่างรอบ ๆ เช่น ' 45 '
});

describe('sumAmount', () => {
  it('รวมยอดได้ถูกต้อง', () => {
    expect(sumAmount([{ amount: 60 }, { amount: 40 }])).toBe(100);
  });

  it('รายการว่างได้ 0', () => {
    expect(sumAmount([])).toBe(0);
  });

  // TODO: เพิ่มกรณีทศนิยม เช่น 0.1 + 0.2 ต้องได้ 0.3
});

ขั้นตอนการปฏิบัติ

  1. สร้างไฟล์ src/utils/validate.test.ts จากโค้ดตั้งต้น รัน npm run test:unit และดูว่าล้มเพราะยังไม่มีไฟล์ validate.ts ถ่ายภาพ

  2. สร้างไฟล์ src/utils/validate.ts เขียนฟังก์ชัน validateAmount(input: string): string และ sumAmount(items) ให้ผ่านทุกการทดสอบ ตามข้อกำหนดที่ทดสอบระบุไว้ (ข้อความผิดพลาดต้องตรงตามที่ทดสอบ) รันซ้ำจนผ่านทั้งหมด

  3. เติมกรณีทดสอบตามคอมเมนต์ TODO 3 ข้อ (ทศนิยมเกิน 2 ตำแหน่ง ตัวเลขที่มีช่องว่างรอบ ๆ และ 0.1 + 0.2) และปรับโค้ดถ้าจำเป็น

  4. ทดลองทำให้ชุดทดสอบล้ม: แก้ validateAmount ให้ผิด 1 จุด (เช่น เปลี่ยน <= 0 เป็น < 0) รันและดูว่ากรณีใดล้ม ถ่ายภาพข้อความ แล้วแก้กลับ

  5. เลือกฟังก์ชันตรรกะของแอปตนเอง อย่างน้อย 2 ฟังก์ชัน (เช่น คำนวณยอด จัดกลุ่ม กรองรายการ ตรวจข้อมูล) ถ้าตรรกะอยู่ในคอมโพเนนต์ ให้แยกออกมาเป็นฟังก์ชันก่อน แล้วออกแบบตารางกรณีทดสอบ 10 กรณี (รหัส รายการที่ทดสอบ ข้อมูลนำเข้า ผลที่คาดหวัง ผลจริง ผ่าน/ไม่ผ่าน) ครอบคลุมการแบ่งกลุ่ม ค่าขอบเขต และกรณีผิดปกติ

  6. เขียนชุดทดสอบด้วย Vitest ตามตารางในข้อ 5 อย่างน้อย 8 กรณี รันให้ผ่าน และเติมคอลัมน์ผลจริงในตาราง

  7. ทดสอบด้วยมือ: ใช้แอปตาม use case ที่เลือก 4 ข้อ (จากใบงาน 2.2) บนเบราว์เซอร์และในโหมดจำลองมือถือ บันทึกผล

  8. เขียนรายงานข้อผิดพลาดที่พบจริงอย่างน้อย 2 ข้อ (จากชุดทดสอบหรือการทดสอบด้วยมือ) ให้ครบองค์ประกอบตามใบความรู้ ถ้าไม่พบข้อผิดพลาดเลย ให้เขียนว่าเหตุใดจึงเชื่อว่ากรณีทดสอบครอบคลุมพอ และเสนอกรณีทดสอบเพิ่ม 2 ข้อ

คำแนะนำ

  • ใช้ npm run test:unit:watch ระหว่างเขียน ชุดทดสอบจะรันซ้ำให้ทุกครั้งที่บันทึกไฟล์
  • ข้อความผิดพลาดของ expect บอกค่าที่คาดหวังกับค่าจริงเสมอ ให้อ่านก่อนแก้
  • ฟังก์ชันที่ทดสอบง่ายคือฟังก์ชันที่ไม่แตะหน้าจอ ฐานข้อมูล หรือเครือข่ายโดยตรง ถ้าทดสอบยาก ให้แยกส่วนที่เป็นตรรกะออกมา
  • ทดสอบเฉพาะสิ่งที่ผู้ใช้หรือโปรแกรมอื่นพึ่งพา ไม่ต้องทดสอบรายละเอียดภายในที่เปลี่ยนบ่อย

สิ่งที่ต้องส่ง

ลิงก์ GitHub (มีไฟล์ validate.test.ts และชุดทดสอบของแอปตนเอง) + ภาพผลรัน npm run test:unit ทั้งกรณีผ่านและกรณีล้มโดยตั้งใจ + ตารางกรณีทดสอบ 10 กรณี + ผลทดสอบด้วยมือ + รายงานข้อผิดพลาด (ส่งตามช่องทางที่ครูกำหนด)
