จุดประสงค์

อ่านตำแหน่งปัจจุบันด้วย Geolocation จัดการกรณีปฏิเสธสิทธิ์ และคำนวณระยะทางถึงสถานที่ที่กำหนดได้

คำชี้แจง

ionic serve ต้องรันอยู่ และเปิดเบราว์เซอร์ที่ localhost (ตำแหน่งและกล้องบนเบราว์เซอร์ใช้ได้เฉพาะ localhost หรือ HTTPS เท่านั้น) ทุกขั้นที่ "วางโค้ดทั้งไฟล์" ให้ลบของเดิมก่อนด้วย Ctrl+A แล้ว Delete ใบงานนี้สร้างหน้า /where ที่อ่านตำแหน่งปัจจุบันและบอกระยะทางถึงสถานที่ที่กำหนด

สิ่งที่ต้องเตรียม

  • ผลงานใบงาน 4.4 หรือโปรเจกต์ Ionic React ที่รัน ionic serve ได้
  • ใบความรู้ที่ 5 หัวข้อสิทธิ์และ Geolocation

ขั้นตอนการปฏิบัติ

  1. เปิดเทอร์มินัลใหม่ (กดปุ่ม + ที่แผงเทอร์มินัล ปล่อยเทอร์มินัลที่รัน ionic serve ไว้) แล้วติดตั้งปลั๊กอินตำแหน่ง

    npm install @capacitor/geolocation

  2. สร้างไฟล์หน้าใหม่ src/pages/Where.tsx (คลิกขวาที่โฟลเดอร์ pages เลือก New File) วางโค้ดนี้ทั้งไฟล์ กด Ctrl+S

    import { useState } from 'react';
import { IonButton, IonContent, IonHeader, IonItem, IonLabel, IonList, IonPage, IonTitle, IonToolbar } from '@ionic/react';
import { Geolocation } from '@capacitor/geolocation';

interface Point { lat: number; lng: number }

// TODO: เปลี่ยนเป็นพิกัดสถานที่ที่ต้องการ (ค่านี้เป็นค่าประมาณกลางเมืองอุดรธานี ให้ค้นพิกัดจริงจาก Google Maps)
const TARGET: Point = { lat: 17.41, lng: 102.79 };
const TARGET_NAME = 'จุดอ้างอิง';

// ระยะทางระหว่างสองพิกัด (กิโลเมตร) ด้วยสูตร Haversine
function distanceKm(a: Point, b: Point): number {
  const R = 6371;
  const toRad = (d: number) => (d * Math.PI) / 180;
  const dLat = toRad(b.lat - a.lat);
  const dLng = toRad(b.lng - a.lng);
  const h =
    Math.sin(dLat / 2) ** 2 +
    Math.cos(toRad(a.lat)) * Math.cos(toRad(b.lat)) * Math.sin(dLng / 2) ** 2;
  return 2 * R * Math.asin(Math.sqrt(h));
}

type Status = 'idle' | 'loading' | 'ok' | 'denied' | 'error';

const Where: React.FC = () => {
  const [status, setStatus] = useState<Status>('idle');
  const [pos, setPos] = useState<Point | null>(null);
  const [accuracy, setAccuracy] = useState(0);

  async function locate() {
    setStatus('loading');
    try {
      const perm = await Geolocation.checkPermissions();
      if (perm.location === 'denied') {
        setStatus('denied');
        return;
      }
      const p = await Geolocation.getCurrentPosition({ enableHighAccuracy: true, timeout: 10000 });
      setPos({ lat: p.coords.latitude, lng: p.coords.longitude });
      setAccuracy(p.coords.accuracy);
      setStatus('ok');
    } catch (e) {
      // TODO: แยกกรณีผู้ใช้กดไม่อนุญาตกับกรณีอื่น ๆ ให้ข้อความต่างกัน
      setStatus('error');
    }
  }

  return (
    <IonPage>
      <IonHeader>
        <IonToolbar color="primary">
          <IonTitle>ตำแหน่งของฉัน</IonTitle>
        </IonToolbar>
      </IonHeader>
      <IonContent className="ion-padding">
        <IonButton expand="block" onClick={locate} disabled={status === 'loading'}>
          {status === 'loading' ? 'กำลังอ่านตำแหน่ง...' : 'ดูตำแหน่งของฉัน'}
        </IonButton>

        {status === 'denied' && <p>ไม่ได้รับสิทธิ์ตำแหน่ง เปิดสิทธิ์ได้ในการตั้งค่าของเบราว์เซอร์หรือเครื่อง</p>}
        {status === 'error' && <p>อ่านตำแหน่งไม่สำเร็จ ลองใหม่อีกครั้ง</p>}

        {status === 'ok' && pos && (
          <IonList>
            <IonItem>
              <IonLabel>
                <h2>พิกัดของคุณ</h2>
                <p>{pos.lat.toFixed(5)}, {pos.lng.toFixed(5)} (ผิดพลาดได้ประมาณ {Math.round(accuracy)} ม.)</p>
              </IonLabel>
            </IonItem>
            <IonItem>
              <IonLabel>
                <h2>ระยะทางถึง{TARGET_NAME}</h2>
                <p>{distanceKm(pos, TARGET).toFixed(2)} กม.</p>
              </IonLabel>
            </IonItem>
          </IonList>
        )}
      </IonContent>
    </IonPage>
  );
};

export default Where;

  5. กดปุ่ม ดูตำแหน่งของฉัน

    ผลที่ควรเห็น: เบราว์เซอร์ถามสิทธิ์ใช้ตำแหน่ง (หน้าต่างเล็กใต้แถบที่อยู่) ถ้าไม่มีหน้าต่างถามและขึ้นข้อความ ไม่ได้รับสิทธิ์ตำแหน่ง แปลว่าเบราว์เซอร์นี้ถูกตั้งเป็นบล็อกไว้ ให้ไปทำข้อ 8 ก่อน

  6. กด อนุญาต (Allow) ในหน้าต่างถามสิทธิ์

    ผลที่ควรเห็น: ปุ่มเปลี่ยนเป็น กำลังอ่านตำแหน่ง... ชั่วครู่ แล้วมีรายการ 2 แถว: พิกัดของคุณ (เลขทศนิยม 5 ตำแหน่ง พร้อมค่าผิดพลาดประมาณ ... ม.) และ ระยะทางถึงจุดอ้างอิง ... กม.

  7. จดค่าที่เห็นลงในช่องบันทึก: พิกัด และค่าความผิดพลาด (accuracy) แล้วตอบว่าทำไมบนคอมพิวเตอร์ค่าผิดพลาดจึงมักมากกว่ามือถือ

    ผลที่ควรเห็น: จดค่าได้ และตอบได้ว่าคอมพิวเตอร์ไม่มี GPS จึงเดาตำแหน่งจากเครือข่าย ทำให้คลาดเคลื่อนมาก

  8. ทดสอบกรณีปฏิเสธสิทธิ์: คลิกไอคอนแม่กุญแจ (หรือไอคอนตั้งค่า) ข้างที่อยู่ในแถบที่อยู่ของ Chrome เลือกการตั้งค่าไซต์ แล้วเปลี่ยน ตำแหน่ง เป็น บล็อก จากนั้นรีเฟรชหน้า (F5) แล้วกดปุ่ม ดูตำแหน่งของฉัน

    ผลที่ควรเห็น: ปุ่มกลับมาเป็นปกติ และมีข้อความ ไม่ได้รับสิทธิ์ตำแหน่ง เปิดสิทธิ์ได้ในการตั้งค่าของเบราว์เซอร์หรือเครื่อง ถ่ายภาพข้อความนี้ แล้วเปลี่ยนสิทธิ์กลับเป็น อนุญาต

  9. ทำให้เป็นสถานที่ของคุณเอง: ค้นหาพิกัดสถานที่ที่เกี่ยวกับแอปของคุณใน Google Maps (คลิกขวาที่จุด แล้วคัดลอกตัวเลขสองตัวแรก คือละติจูดและลองจิจูด) แล้วแก้ 2 บรรทัดนี้ในไฟล์ Where.tsx (ตัวอย่างใช้กรุงเทพมหานคร)

  10. ตรวจว่าตัวเลขสมเหตุสมผล: ใช้ Google Maps วัดระยะตรงจากตำแหน่งของคุณถึงสถานที่นั้น แล้วเทียบกับระยะที่แอปแสดง

  14. ตอบคำถามท้ายใบงานในช่องบันทึก: เหตุใดเปิดแอปบนมือถือผ่าน http://หมายเลข-IP แล้วใช้ตำแหน่งไม่ได้ และจะทดสอบบนมือถือได้อย่างไร (ใบความรู้ที่ 5 กล่องข้อควรรู้)

    ผลที่ควรเห็น: ตอบได้ว่าเบราว์เซอร์อนุญาตตำแหน่งเฉพาะ HTTPS หรือ localhost ทางทดสอบคือเว็บแบบ HTTPS หรือสร้างแอป Android

คำแนะนำ

  • ถ้าไม่ขึ้นหน้าต่างขอสิทธิ์ ตรวจว่าเปิดที่ localhost ไม่ใช่หมายเลข IP และดูไอคอนที่แถบที่อยู่ของเบราว์เซอร์
  • บนคอมพิวเตอร์ ตำแหน่งอาจคลาดเคลื่อนหลายร้อยเมตรถึงหลายกิโลเมตร เพราะไม่มี GPS ให้ดูค่า accuracy ประกอบ
  • พิกัดละติจูดอยู่ก่อนลองจิจูดเสมอ (lat, lng) ถ้าสลับกัน ระยะทางจะผิดมาก
