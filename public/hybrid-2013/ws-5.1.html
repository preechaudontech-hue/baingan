จุดประสงค์

อ่านตำแหน่งปัจจุบันด้วย Geolocation จัดการกรณีปฏิเสธสิทธิ์ และคำนวณระยะทางถึงสถานที่ที่กำหนดได้

คำชี้แจง

สร้างหน้าที่อ่านตำแหน่งของอุปกรณ์แล้วแสดงระยะทางไปยังสถานที่หนึ่งแห่ง ทดสอบบนคอมพิวเตอร์ที่ localhost ก่อน ซึ่งเบราว์เซอร์จะถามสิทธิ์ตำแหน่ง

สิ่งที่ต้องเตรียม

  • ผลงานใบงาน 4.4 หรือโปรเจกต์ Ionic React ที่รัน ionic serve ได้
  • ใบความรู้ที่ 5 หัวข้อสิทธิ์และ Geolocation

โค้ดตั้งต้น

คัดลอกไปวางในไฟล์ที่ระบุในโค้ด
import { useState } from 'react';
import { IonButton, IonContent, IonHeader, IonItem, IonLabel, IonList, IonPage, IonTitle, IonToolbar } from '@ionic/react';
import { Geolocation } from '@capacitor/geolocation';

interface Point { lat: number; lng: number }

// TODO: เปลี่ยนเป็นพิกัดสถานที่ที่ต้องการ (ค่านี้เป็นค่าประมาณกลางเมืองอุดรธานี ให้ค้นพิกัดจริงจาก Google Maps)
const TARGET: Point = { lat: 17.41, lng: 102.79 };
const TARGET_NAME = 'จุดอ้างอิง';

// ระยะทางระหว่างสองพิกัด (กิโลเมตร) ด้วยสูตร Haversine
function distanceKm(a: Point, b: Point): number {
  const R = 6371;
  const toRad = (d: number) => (d * Math.PI) / 180;
  const dLat = toRad(b.lat - a.lat);
  const dLng = toRad(b.lng - a.lng);
  const h =
    Math.sin(dLat / 2) ** 2 +
    Math.cos(toRad(a.lat)) * Math.cos(toRad(b.lat)) * Math.sin(dLng / 2) ** 2;
  return 2 * R * Math.asin(Math.sqrt(h));
}

type Status = 'idle' | 'loading' | 'ok' | 'denied' | 'error';

const Where: React.FC = () => {
  const [status, setStatus] = useState<Status>('idle');
  const [pos, setPos] = useState<Point | null>(null);
  const [accuracy, setAccuracy] = useState(0);

  async function locate() {
    setStatus('loading');
    try {
      const perm = await Geolocation.checkPermissions();
      if (perm.location === 'denied') {
        setStatus('denied');
        return;
      }
      const p = await Geolocation.getCurrentPosition({ enableHighAccuracy: true, timeout: 10000 });
      setPos({ lat: p.coords.latitude, lng: p.coords.longitude });
      setAccuracy(p.coords.accuracy);
      setStatus('ok');
    } catch (e) {
      // TODO: แยกกรณีผู้ใช้กดไม่อนุญาตกับกรณีอื่น ๆ ให้ข้อความต่างกัน
      setStatus('error');
    }
  }

  return (
    <IonPage>
      <IonHeader>
        <IonToolbar color="primary">
          <IonTitle>ตำแหน่งของฉัน</IonTitle>
        </IonToolbar>
      </IonHeader>
      <IonContent className="ion-padding">
        <IonButton expand="block" onClick={locate} disabled={status === 'loading'}>
          {status === 'loading' ? 'กำลังอ่านตำแหน่ง...' : 'ดูตำแหน่งของฉัน'}
        </IonButton>

        {status === 'denied' && <p>ไม่ได้รับสิทธิ์ตำแหน่ง เปิดสิทธิ์ได้ในการตั้งค่าของเบราว์เซอร์หรือเครื่อง</p>}
        {status === 'error' && <p>อ่านตำแหน่งไม่สำเร็จ ลองใหม่อีกครั้ง</p>}

        {status === 'ok' && pos && (
          <IonList>
            <IonItem>
              <IonLabel>
                <h2>พิกัดของคุณ</h2>
                <p>{pos.lat.toFixed(5)}, {pos.lng.toFixed(5)} (ผิดพลาดได้ประมาณ {Math.round(accuracy)} ม.)</p>
              </IonLabel>
            </IonItem>
            <IonItem>
              <IonLabel>
                <h2>ระยะทางถึง{TARGET_NAME}</h2>
                <p>{distanceKm(pos, TARGET).toFixed(2)} กม.</p>
              </IonLabel>
            </IonItem>
          </IonList>
        )}
      </IonContent>
    </IonPage>
  );
};

export default Where;

ขั้นตอนการปฏิบัติ

  1. ติดตั้งปลั๊กอินด้วยคำสั่ง npm install @capacitor/geolocation แล้วรีสตาร์ท ionic serve ถ้าจำเป็น

  2. สร้างไฟล์หน้าใหม่จากโค้ดตั้งต้น เปิดที่ localhost แล้วกดปุ่ม ดูว่าเบราว์เซอร์ถามสิทธิ์ตำแหน่ง ลองกด อนุญาต และตรวจว่าแสดงพิกัดและระยะทาง

  3. เปลี่ยน TARGET และ TARGET_NAME เป็นสถานที่ที่เกี่ยวกับแอปของตนเอง (ค้นพิกัดจาก Google Maps) และตรวจว่าระยะทางสมเหตุสมผล

  4. ทดสอบกรณีปฏิเสธ: เข้าไปตั้งค่าสิทธิ์ตำแหน่งของเว็บไซต์ในเบราว์เซอร์ให้เป็น บล็อก แล้วกดปุ่มอีกครั้ง ดูว่าแอปแสดงข้อความอธิบายและยังใช้งานต่อได้ ถ่ายภาพ

  5. ปรับปรุงการจัดการข้อผิดพลาดตามคอมเมนต์ TODO ให้ข้อความของแต่ละกรณีบอกผู้ใช้ว่าต้องทำอะไรต่อ

  6. เขียนฟังก์ชันทดสอบ distanceKm ด้วยค่าที่รู้คำตอบ (จุดเดียวกันต้องได้ 0 กม. และจุด 0,0 กับ 0,1 องศาต้องได้ประมาณ 111 กม.) ตรวจใน Console แล้วบันทึกผล

  7. ตอบคำถาม: เหตุใดเปิดแอปบนมือถือผ่าน http://หมายเลข-IP แล้วใช้ตำแหน่งไม่ได้ และจะทดสอบบนมือถือได้อย่างไร (อ้างอิงใบความรู้)

คำแนะนำ

  • ถ้าไม่ขึ้นหน้าต่างขอสิทธิ์ ให้ตรวจว่าเปิดที่ localhost ไม่ใช่หมายเลข IP และดูไอคอนที่แถบที่อยู่ของเบราว์เซอร์
  • บนคอมพิวเตอร์ ตำแหน่งที่ได้อาจคลาดเคลื่อนหลายร้อยเมตรถึงหลายกิโลเมตร เพราะไม่มี GPS ข้อมูลนี้ปกติ ให้ดูค่า accuracy ประกอบ
  • พิกัดละติจูดอยู่ก่อนลองจิจูดเสมอ (lat, lng) ถ้าสลับกันระยะทางจะผิดมาก

สิ่งที่ต้องส่ง

ภาพหน้าจอ 3 ภาพ (ก่อนกด แสดงตำแหน่งและระยะทาง และกรณีปฏิเสธสิทธิ์) + ผลทดสอบฟังก์ชัน distanceKm + คำตอบข้อ 7 + ลิงก์ GitHub (ส่งตามช่องทางที่ครูกำหนด)
