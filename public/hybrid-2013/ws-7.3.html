ใบงานที่ 7.3 เชื่อมแอปของฉันกับ API และเก็บสำเนาในเครื่อง

เวลา 180 นาที · งานเดี่ยว · คะแนนเต็ม 10

จุดประสงค์

เชื่อมแอปของตนเองกับ API ภายนอกอย่างน้อย 1 รายการ และเก็บสำเนาข้อมูลในเครื่องเพื่อใช้ตอนออฟไลน์ พร้อมแสดงเวลาที่อัปเดตล่าสุด

คำชี้แจง

เลือก API ที่เกี่ยวข้องกับแอปของตนเอง แล้วเชื่อมเข้ากับแอป โดยใช้ฟังก์ชัน fetchWithCache ที่ให้มาเพื่อเก็บสำเนา ถ้าครูจัด API ของวิชาไว้ ให้ใช้ API นั้นแทน

สิ่งที่ต้องเตรียม

  • ผลงานใบงาน 6.3 และ 7.1 - 7.2
  • ใบความรู้ที่ 7 หัวข้อเครือข่ายที่ไม่แน่นอน และการอ่านเอกสาร API

โค้ดตั้งต้น

คัดลอกไปวางในไฟล์ที่ระบุในโค้ด
// src/api/cache.ts
import { Preferences } from '@capacitor/preferences';

interface Cached<T> { savedAt: number; data: T }

export interface Loaded<T> { data: T; fromCache: boolean; savedAt: number }

// โหลดจากเครือข่าย ถ้าล้มเหลวใช้สำเนาที่เก็บไว้ (ถ้าไม่มีสำเนา โยนข้อผิดพลาดเดิม)
export async function fetchWithCache<T>(key: string, load: () => Promise<T>): Promise<Loaded<T>> {
  try {
    const data = await load();
    const savedAt = Date.now();
    await Preferences.set({ key, value: JSON.stringify({ savedAt, data } as Cached<T>) });
    return { data, fromCache: false, savedAt };
  } catch (e) {
    const { value } = await Preferences.get({ key });
    if (value) {
      try {
        const c = JSON.parse(value) as Cached<T>;
        return { data: c.data, fromCache: true, savedAt: c.savedAt };
      } catch {
        // สำเนาเสียหาย ถือว่าไม่มีสำเนา
      }
    }
    throw e;
  }
}

// TODO: เขียนฟังก์ชันแปลงเวลา savedAt เป็นข้อความ เช่น "อัปเดตเมื่อ 5 นาทีที่แล้ว" หรือ "อัปเดตเมื่อ 14:30"
export function timeAgo(savedAt: number): string {
  return String(savedAt);
}

ขั้นตอนการปฏิบัติ

  1. เลือก API ที่เกี่ยวข้องกับแอปของตนเอง 1 รายการ ตัวเลือกที่ใช้ได้โดยไม่ต้องสมัคร: Open-Meteo (สภาพอากาศ) PokeAPI (pokeapi.co/api/v2/pokemon ข้อมูลโปเกมอน) หรือ Open Library (openlibrary.org/search.json หนังสือ) หรือ API ที่ครูกำหนด แล้วเขียนสรุปเอกสาร API ตามตารางในใบความรู้ (Base URL, endpoint ที่ใช้, พารามิเตอร์, ตัวอย่างคำตอบ, การยืนยันตัวตน, ขีดจำกัด, เงื่อนไขการใช้งาน)

  2. ทดสอบเรียก API ด้วยเบราว์เซอร์หรือ DevTools ก่อนเขียนโค้ด แล้วกำหนด interface ของคำตอบ โดยระบุฟิลด์ที่อาจไม่มีเป็น optional

  3. สร้างไฟล์ src/api/cache.ts จากโค้ดตั้งต้น เขียนฟังก์ชัน timeAgo ให้แสดงเวลาที่อัปเดตล่าสุดเป็นภาษาไทย

  4. สร้างฟังก์ชันเรียก API ของแอปตนเองด้วย getJson และห่อด้วย fetchWithCache ใช้ในหน้าจอ แสดงสถานะ กำลังโหลด ผิดพลาด ว่าง และข้อความ "กำลังแสดงข้อมูลเก่า อัปเดตเมื่อ ..." เมื่อ fromCache เป็น true

  5. เพิ่มปุ่มโหลดใหม่ และป้องกันการกดซ้ำขณะโหลด ยกเลิกคำขอเมื่อออกจากหน้า

  6. ทดสอบ 3 สถานการณ์: ออนไลน์ครั้งแรก, ออนไลน์ครั้งที่สอง, และออฟไลน์ (ใช้ Offline ใน DevTools) แล้วรีเฟรชต้องยังเห็นข้อมูลเดิมพร้อมข้อความ ถ่ายภาพทุกสถานการณ์ และทดสอบกรณีออฟไลน์ตั้งแต่ยังไม่เคยโหลด (ต้องแจ้งข้อผิดพลาดที่เข้าใจได้)

  7. ตรวจความปลอดภัยของแอป: ตรวจว่าไม่มีกุญแจลับในโค้ด ใช้ HTTPS ทุกที่ และเขียนรายการข้อมูลที่แอปส่งออกไปยัง API (เช่น พิกัด ชื่อเมือง) พร้อมเหตุผลที่จำเป็น

  8. เขียนสรุปสิ่งที่ยังไม่สมบูรณ์ และแผนแก้ไข 3 ข้อ

คำแนะนำ

  • ใช้ key สำหรับเก็บสำเนาที่แยกตามพารามิเตอร์ของคำขอ เช่น weather-อุดรธานี ไม่เช่นนั้นข้อมูลของเมืองหนึ่งจะไปทับของอีกเมือง
  • ถ้า API ที่เลือกไม่อนุญาต CORS ให้เปลี่ยนไปเลือก API อื่น หรือปรึกษาครู
  • แสดงข้อมูลจากสำเนาด้วยความระมัดระวัง ต้องบอกผู้ใช้ชัดเจนว่าเป็นข้อมูลเก่า

สิ่งที่ต้องส่ง

ลิงก์ GitHub + สรุปเอกสาร API + ภาพหน้าจอ 3 สถานการณ์ (ออนไลน์ ข้อมูลเก่าตอนออฟไลน์ และออฟไลน์ครั้งแรก) + รายการข้อมูลที่ส่งออกและเหตุผล + สิ่งที่ยังไม่สมบูรณ์และแผนแก้ไข (ส่งตามช่องทางที่ครูกำหนด)
