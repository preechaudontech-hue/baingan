จุดประสงค์

สร้างฐานข้อมูล SQLite ในแอปและทำ CRUD ครบด้วยคำสั่ง SQL ที่ปลอดภัย ข้อมูลคงอยู่หลังรีเฟรช

คำชี้แจง

ionic serve ต้องรันอยู่ ทุกขั้นที่ "วางโค้ดทั้งไฟล์" ให้ลบของเดิมก่อนด้วย Ctrl+A แล้ว Delete ข้อมูลที่เก็บเป็นของเบราว์เซอร์ที่ใช้ทดสอบ (ผูกกับที่อยู่ localhost:พอร์ต) ใบงานนี้สร้างฐานข้อมูล SQLite ในแอป (ใช้ sql.js) ทำ CRUD และทดสอบผ่านหน้าทดสอบที่ให้มา แล้วนำไปต่อกับหน้ารายการจากใบงาน 4.3

export interface Item {

  category: string;
  amount: number;

function toBase64(bytes: Uint8Array): string {
  let s = '';
  for (let i = 0; i < bytes.length; i += 0x8000) {
    s += String.fromCharCode(...bytes.subarray(i, i + 0x8000));
  }
  return btoa(s);
}

function fromBase64(text: string): Uint8Array {
  const s = atob(text);
  const bytes = new Uint8Array(s.length);
  for (let i = 0; i < s.length; i++) bytes[i] = s.charCodeAt(i);
  return bytes;
}

export async function getDb(): Promise<Database> {
  if (db) return db;
  const SQL = await initSqlJs({ locateFile: () => wasmUrl });
  const saved = await Preferences.get({ key: STORAGE_KEY });
  db = saved.value ? new SQL.Database(fromBase64(saved.value)) : new SQL.Database();
  db.run(`
    CREATE TABLE IF NOT EXISTS items (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      title TEXT NOT NULL,
      category TEXT NOT NULL,
      amount REAL NOT NULL CHECK (amount > 0)
    )
  `);
  return db;
}

async function persist(): Promise<void> {
  const conn = await getDb();
  await Preferences.set({ key: STORAGE_KEY, value: toBase64(conn.export()) });
}

export async function listItems(): Promise<Item[]> {
  const conn = await getDb();
  const stmt = conn.prepare('SELECT id, title, category, amount FROM items ORDER BY id DESC');
  const rows: Item[] = [];
  while (stmt.step()) rows.push(stmt.getAsObject() as unknown as Item);
  stmt.free();
  return rows;
}

export async function searchUnsafe(text: string): Promise<Item[]> {
  const conn = await getDb();
  const res = conn.exec("SELECT id, title, category, amount FROM items WHERE title = '" + text + "'");
  if (res.length === 0) return [];
  return res[0].values.map(v => ({ id: Number(v[0]), title: String(v[1]), category: String(v[2]), amount: Number(v[3]) }));
}

// ปลอดภัย: ใช้เครื่องหมาย ? แล้วส่งค่าแยกเป็นพารามิเตอร์
export async function searchSafe(text: string): Promise<Item[]> {
  const conn = await getDb();
  const stmt = conn.prepare('SELECT id, title, category, amount FROM items WHERE title = ?');
  stmt.bind([text]);
  const rows: Item[] = [];
  while (stmt.step()) rows.push(stmt.getAsObject() as unknown as Item);
  stmt.free();
  return rows;
}

import { IonButton, IonContent, IonHeader, IonInput, IonItem, IonLabel, IonList, IonPage, IonTitle, IonToolbar } from '@ionic/react';
import {
  addItem, deleteItem, listItems, searchSafe, searchUnsafe, totalsByCategory, updateItem,
  CategoryTotal, Item,
} from '../db/database';

const DbTest: React.FC = () => {
  const [items, setItems] = useState<Item[]>([]);
  const [totals, setTotals] = useState<CategoryTotal[]>([]);
  const [message, setMessage] = useState('');
  const [text, setText] = useState('');

  async function reload() {
    setItems(await listItems());
    setTotals(await totalsByCategory());
  }

  useEffect(() => {
    reload().catch(() => setMessage('อ่านฐานข้อมูลไม่สำเร็จ'));
  }, []);

  async function run(job: () => Promise<void>) {
    try {
      await job();
      setMessage('');
      await reload();
    } catch (e) {
      setMessage('ผิดพลาด: ' + String(e));
    }
  }

  return (
    <IonPage>
      <IonHeader>
        <IonToolbar color="primary">
          <IonTitle>ทดสอบฐานข้อมูล ({items.length})</IonTitle>

      </IonHeader>
      <IonContent className="ion-padding">
        <IonButton onClick={() => run(() => addItem('ข้าวมันไก่', 'อาหาร', 60))}>เพิ่ม</IonButton>
        <IonButton onClick={() => run(() => addItem('ผิดกฎ', 'อาหาร', 0))}>เพิ่มจำนวน 0</IonButton>

        <IonButton color="danger" onClick={() => run(async () => { if (items[0]) await deleteItem(items[0].id); })}>ลบรายการแรก</IonButton>

        <IonInput label="ค้นหาชื่อ" labelPlacement="floating" fill="outline" value={text} onIonInput={e => setText(String(e.detail.value ?? ''))} />
        <IonButton onClick={async () => setMessage('ไม่ปลอดภัย: ' + (await searchUnsafe(text)).length + ' แถว')}>ค้นแบบต่อสตริง</IonButton>
        <IonButton onClick={async () => setMessage('ปลอดภัย: ' + (await searchSafe(text)).length + ' แถว')}>ค้นแบบพารามิเตอร์</IonButton>

    ผลที่ควรเห็น: ขึ้นข้อความ ไม่ปลอดภัย: N แถว โดย N เท่ากับจำนวนรายการทั้งหมด (ทั้งที่ไม่มีรายการชื่อแบบนั้น แปลว่าข้อความที่พิมพ์ไปเปลี่ยนความหมายของคำสั่ง SQL)

  17. ทดสอบข้อ 2: ข้อความเดิมในช่อง แต่กดปุ่ม ค้นแบบพารามิเตอร์

    ผลที่ควรเห็น: ขึ้นข้อความ ปลอดภัย: 0 แถว (ข้อความถูกมองเป็นชื่อที่ต้องการค้น ไม่ใช่ส่วนของคำสั่ง) ถ่ายภาพผลทั้งสองข้อ

    ผลที่ควรเห็น: ขึ้น ปลอดภัย: จำนวนแถวที่ชื่อ ข้าวมันไก่ ตรงตัว

คำแนะนำ

  • ทุกฟังก์ชันที่แก้ข้อมูลต้องเรียก persist() ถ้าลืม ข้อมูลจะหายเมื่อรีเฟรช
  • ใช้ conn.run สำหรับคำสั่งที่ไม่คืนแถว และ prepare, step, getAsObject, free สำหรับ SELECT
  • ถ้าข้อมูลทดสอบเสียหาย ล้างได้โดยลบคีย์ CapacitorStorage.myapp-db (หรือชื่อที่คุณตั้ง) ใน Local Storage ผ่านแท็บ Application ของ DevTools
