โค้ดตั้งต้น

คัดลอกไปวางในไฟล์ที่ระบุในโค้ด
import initSqlJs, { Database } from 'sql.js';
import wasmUrl from 'sql.js/dist/sql-wasm.wasm?url';
import { Preferences } from '@capacitor/preferences';

// src/db/database.ts
const STORAGE_KEY = 'myapp-db';
let db: Database | null = null;

export interface Item {
  id: number;
  title: string;
  category: string;
  amount: number;
}

function toBase64(bytes: Uint8Array): string {
  let s = '';
  for (let i = 0; i < bytes.length; i += 0x8000) {
    s += String.fromCharCode(...bytes.subarray(i, i + 0x8000));
  }
  return btoa(s);
}

function fromBase64(text: string): Uint8Array {
  const s = atob(text);
  const bytes = new Uint8Array(s.length);
  for (let i = 0; i < s.length; i++) bytes[i] = s.charCodeAt(i);
  return bytes;
}

export async function getDb(): Promise<Database> {
  if (db) return db;
  const SQL = await initSqlJs({ locateFile: () => wasmUrl });
  const saved = await Preferences.get({ key: STORAGE_KEY });
  db = saved.value ? new SQL.Database(fromBase64(saved.value)) : new SQL.Database();
  db.run(`
    CREATE TABLE IF NOT EXISTS items (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      title TEXT NOT NULL,
      category TEXT NOT NULL,
      amount REAL NOT NULL CHECK (amount > 0)
    )
  `);
  return db;
}

async function persist(): Promise<void> {
  const conn = await getDb();
  await Preferences.set({ key: STORAGE_KEY, value: toBase64(conn.export()) });
}

export async function listItems(): Promise<Item[]> {
  const conn = await getDb();
  const stmt = conn.prepare('SELECT id, title, category, amount FROM items ORDER BY id DESC');
  const rows: Item[] = [];
  while (stmt.step()) rows.push(stmt.getAsObject() as unknown as Item);
  stmt.free();
  return rows;
}

export async function addItem(title: string, category: string, amount: number): Promise<void> {
  const conn = await getDb();
  conn.run('INSERT INTO items (title, category, amount) VALUES (?, ?, ?)', [title, category, amount]);
  await persist();
}

ขั้นตอนการปฏิบัติ

  1. ติดตั้งแพ็กเกจด้วยคำสั่ง npm install sql.js @capacitor/preferences และ npm install -D @types/sql.js แล้วสร้างไฟล์ src/db/database.ts จากโค้ดตั้งต้น (ตัดบรรทัดคอมเมนต์ชื่อไฟล์ออกก็ได้) เปลี่ยนค่า STORAGE_KEY เป็นชื่อของแอปตนเอง

  2. เติมฟังก์ชัน updateItem และ deleteItem โดยใช้พารามิเตอร์ ? และเรียก persist() ตามแบบของ addItem

  3. เติมฟังก์ชัน totalsByCategory ด้วยคำสั่ง SELECT ... GROUP BY ตามคอมเมนต์ (ใช้ prepare, step, getAsObject เหมือน listItems)

  4. แก้หน้ารายการจากใบงาน 4.3 ให้โหลดข้อมูลจากฐานข้อมูลด้วย listItems ใน useEffect มีสถานะกำลังโหลดและข้อผิดพลาด เพิ่ม ลบ และแก้รายการผ่านฟังก์ชันเหล่านี้แล้วโหลดรายการใหม่

  5. ทดสอบ: เพิ่ม แก้ ลบ แล้วรีเฟรชหน้า ตรวจว่าข้อมูลยังอยู่ ถ่ายภาพก่อนและหลังรีเฟรช ทดสอบกรณีจำนวนเงินเป็น 0 ให้ CHECK ในฐานข้อมูลปฏิเสธ แล้วดูว่าแอปจับข้อผิดพลาดและแจ้งผู้ใช้อย่างไร

  6. ทดสอบความปลอดภัย: ลองเขียนฟังก์ชันค้นหาแบบต่อสตริง (searchUnsafe) ที่ใช้ title ต่อเข้าในคำสั่ง แล้วค้นด้วยข้อความ ' OR '1'='1 สังเกตผลว่าได้ข้อมูลทุกแถว จากนั้นเขียนแบบใช้พารามิเตอร์ (searchSafe) และทดสอบด้วยข้อความเดียวกัน บันทึกผลทั้งสอง แล้วลบ searchUnsafe ออกจากแอป

  7. เขียนคำสั่ง SQL 5 ข้อที่เกี่ยวกับข้อมูลของแอปตนเอง (เช่น รายการแพงที่สุด 3 อันดับ ยอดรวมต่อหมวด รายการที่ชื่อมีคำว่า ... ) ทดสอบผ่านฟังก์ชันในแอปหรือใน Console และบันทึกผลที่ได้

คำแนะนำ

  • ถ้า TypeScript ฟ้อง import 'sql.js/dist/sql-wasm.wasm?url' ให้ตรวจว่าโปรเจกต์ใช้ Vite และมีไฟล์ vite-env.d.ts ที่อ้างอิง vite/client
  • ทุกฟังก์ชันที่แก้ข้อมูลต้องเรียก persist() ถ้าลืม ข้อมูลจะหายเมื่อรีเฟรช
  • ใช้ db.run สำหรับคำสั่งที่ไม่คืนแถว และ prepare, step, getAsObject, free สำหรับ SELECT
  • ถ้าข้อมูลที่ทดสอบเสียหาย ล้างได้โดยลบคีย์ของฐานข้อมูลใน Local Storage ผ่านแท็บ Application ของ DevTools

สิ่งที่ต้องส่ง

ลิงก์ GitHub + ภาพหน้าจอ (ก่อนและหลังรีเฟรช และกรณี CHECK ปฏิเสธ) + ผลทดสอบ SQL injection ทั้งสองแบบ + คำสั่ง SQL 5 ข้อพร้อมผล (ส่งตามช่องทางที่ครูกำหนด)
