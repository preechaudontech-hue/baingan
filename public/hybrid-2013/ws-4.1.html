ใบงานที่ 4.1 จัดวางหน้าหลักด้วย Ionic ตามแบบร่าง

เวลา 120 นาที · งานเดี่ยว · คะแนนเต็ม 10

จุดประสงค์

สร้างหน้าหลักของแอปตนเองตามแบบร่าง ด้วยคอมโพเนนต์ Ionic IonGrid และธีมสีที่กำหนดเอง

คำชี้แจง

ใบงานนี้ต้องรัน ionic serve ค้างไว้ และเปิดโปรเจกต์ใน VS Code ทุกขั้นที่ "วางโค้ดทั้งไฟล์" ให้คลิกในไฟล์ กด Ctrl+A แล้ว Delete ก่อนวาง ใช้ข้อมูลตัวอย่างก่อน แล้วค่อยเปลี่ยนเป็นข้อมูลของแอปตนเองจาก wireframe ของใบงาน 2.4

สิ่งที่ต้องเตรียม

  • ผลงานใบงาน 2.4 (wireframe หน้าหลัก)
  • ใบความรู้ที่ 4 หัวข้อโครงหน้า Grid และธีมสี
  • โปรเจกต์ Ionic React ที่รัน ionic serve อยู่

ขั้นตอนการปฏิบัติ

  1. เตรียมความพร้อม: เปิดโปรเจกต์ใน VS Code เปิดเทอร์มินัลแล้วรัน ionic serve เปิดหน้าเว็บใน Chrome แล้วกด Ctrl+Shift+M เพื่อดูแบบจอมือถือ จากนั้นเปิดไฟล์ src/pages/Home.tsx

    ผลที่ควรเห็น: เห็นหน้าแอปบนเบราว์เซอร์และเห็นไฟล์ Home.tsx ใน VS Code

  2. ลบทั้งไฟล์ Home.tsx แล้ววางโค้ดตั้งต้นนี้ กด Ctrl+S

    ไฟล์ src/pages/Home.tsx
    import {
  IonCard, IonCardContent, IonCardHeader, IonCardSubtitle, IonCardTitle, IonCol, IonContent, IonFab,
  IonFabButton, IonGrid, IonHeader, IonIcon, IonPage, IonRow, IonTitle, IonToolbar,
} from '@ionic/react';
import { add } from 'ionicons/icons';

interface Item { id: number; title: string; category: string; amount: number }

// TODO: เปลี่ยนเป็นข้อมูลตัวอย่างของแอปตนเอง
const ITEMS: Item[] = [
  { id: 1, title: 'ข้าวกลางวัน', category: 'อาหาร', amount: 60 },
  { id: 2, title: 'ค่ารถ', category: 'เดินทาง', amount: 40 },
  { id: 3, title: 'ซื้อสมุด', category: 'การเรียน', amount: 85 },
  { id: 4, title: 'ชานมไข่มุก', category: 'อาหาร', amount: 55 },
];

const Home: React.FC = () => {
  const total = ITEMS.reduce((sum, i) => sum + i.amount, 0);

  return (
    <IonPage>
      <IonHeader>
        <IonToolbar color="primary">
          <IonTitle>ชื่อแอปของฉัน</IonTitle>
        </IonToolbar>
      </IonHeader>
      <IonContent className="ion-padding">
        <IonCard>
          <IonCardHeader>
            <IonCardSubtitle>ยอดรวมเดือนนี้</IonCardSubtitle>
            <IonCardTitle>{total} บาท</IonCardTitle>
          </IonCardHeader>
        </IonCard>

        <IonGrid>
          <IonRow>
            {ITEMS.map(i => (
              <IonCol size="6" key={i.id}>
                <IonCard>
                  <IonCardHeader>
                    <IonCardSubtitle>{i.category}</IonCardSubtitle>
                    <IonCardTitle>{i.title}</IonCardTitle>
                  </IonCardHeader>
                  <IonCardContent>{i.amount} บาท</IonCardContent>
                </IonCard>
              </IonCol>
            ))}
          </IonRow>
        </IonGrid>

        <IonFab slot="fixed" vertical="bottom" horizontal="end">
          <IonFabButton>
            <IonIcon icon={add} />
          </IonFabButton>
        </IonFab>
      </IonContent>
    </IonPage>
  );
};

export default Home;
    ผลที่ควรเห็น: แถบหัวสีน้ำเงินเขียนว่า ชื่อแอปของฉัน การ์ดยอดรวม 240 บาท การ์ด 4 ใบเรียง 2 คอลัมน์ และปุ่มกลมเครื่องหมาย + ที่มุมขวาล่าง

  3. อ่านโครงของหน้า: ดูว่ามี IonPage > IonHeader (แถบหัว) และ IonContent (เนื้อหา) แล้วเขียนในช่องบันทึกว่า IonCard, IonGrid, IonFab แต่ละอย่างคือส่วนไหนบนหน้าจอ

    ผลที่ควรเห็น: ตอบได้ว่า IonCard คือการ์ดสี่เหลี่ยม IonGrid คือตารางแถวและคอลัมน์ที่จัดการ์ด 4 ใบ IonFab คือปุ่มลอยมุมขวาล่าง

  4. เปลี่ยนชื่อแอปบนแถบหัว: ในบรรทัด <IonTitle> เปลี่ยนข้อความเป็นชื่อแอปของคุณ (ตัวอย่างใช้ MyExpense) กด Ctrl+S

    ไฟล์ src/pages/Home.tsx
    <IonTitle>MyExpense</IonTitle>
    ผลที่ควรเห็น: แถบหัวเปลี่ยนเป็นชื่อใหม่ทันที

  5. เปลี่ยนข้อมูลตัวอย่างรายการแรก: หาบรรทัด { id: 1, title: 'ข้าวกลางวัน', ... } ในอาเรย์ ITEMS เปลี่ยนเป็นบรรทัดนี้ กด Ctrl+S

    ไฟล์ src/pages/Home.tsx
    { id: 1, title: 'ข้าวมันไก่', category: 'อาหาร', amount: 65 },
    ผลที่ควรเห็น: การ์ดแรกเปลี่ยนเป็น ข้าวมันไก่ 65 บาท และยอดรวมด้านบนเปลี่ยนจาก 240 เป็น 245 บาท

  6. เพิ่มรายการที่ 5: ต่อท้ายบรรทัดของ ชานมไข่มุก ให้เพิ่มบรรทัดใหม่ตามนี้ (เปลี่ยนเป็นข้อมูลของแอปคุณได้) กด Ctrl+S

    ไฟล์ src/pages/Home.tsx
    { id: 4, title: 'ชานมไข่มุก', category: 'อาหาร', amount: 55 },
  { id: 5, title: 'ค่าอินเทอร์เน็ต', category: 'อื่น ๆ', amount: 100 },
    ผลที่ควรเห็น: การ์ดที่ 5 ปรากฏเป็นการ์ดเดียวในแถวสุดท้าย และยอดรวมเป็น 345 บาท

  7. ลองเปลี่ยนความกว้างคอลัมน์: หาบรรทัด <IonCol size="6" key={i.id}> เปลี่ยน 6 เป็น 12 กด Ctrl+S (Grid แบ่งแถวเป็น 12 ส่วน ตัวเลขคือจำนวนส่วนที่ใช้)

    ไฟล์ src/pages/Home.tsx
    <IonCol size="12" key={i.id}>
    ผลที่ควรเห็น: การ์ดทุกใบกว้างเต็มแถว เรียงคอลัมน์เดียว

  8. แก้เป็นแบบปรับตามขนาดจอ: เปลี่ยนบรรทัดเดิมเป็นบรรทัดนี้ กด Ctrl+S (บนจอมือถือใช้ 2 คอลัมน์ บนจอกว้างตั้งแต่ 768 พิกเซลขึ้นไปใช้ 3 คอลัมน์)

    ไฟล์ src/pages/Home.tsx
    <IonCol size="6" sizeMd="4" key={i.id}>
    ผลที่ควรเห็น: บนโหมดมือถือกลับเป็น 2 คอลัมน์ ลองเลือกอุปกรณ์ iPad ในแถบ Device toolbar ของ Chrome จะเห็น 3 คอลัมน์

  9. เปลี่ยนสีหลักของแอป: เปิดไฟล์ src/theme/variables.css (มีแต่คอมเมนต์สองบรรทัด) เพิ่มโค้ดนี้ต่อท้ายไฟล์ แล้วกด Ctrl+S (สีหนึ่งสีมีตัวแปร 6 ตัว ต้องใส่ให้ครบ ใช้เครื่องมือ Color Generator ในเอกสาร Ionic สร้างชุดสีที่ชอบได้)

    ไฟล์ src/theme/variables.css
    :root {
  --ion-color-primary: #0f6b5c;
  --ion-color-primary-rgb: 15, 107, 92;
  --ion-color-primary-contrast: #ffffff;
  --ion-color-primary-contrast-rgb: 255, 255, 255;
  --ion-color-primary-shade: #0d5e51;
  --ion-color-primary-tint: #277a6c;
}
    ผลที่ควรเห็น: แถบหัวและปุ่มกลมที่มุมขวาล่างเปลี่ยนเป็นสีเขียวเข้ม โดยไม่ต้องแก้ไฟล์ Home.tsx

  10. ทดลองสีอื่น: เปลี่ยนค่า #0f6b5c และ #0d5e51 และ #277a6c เป็นสีอื่นของคุณ (ลองกับ Color Generator) แล้วบันทึก

    ผลที่ควรเห็น: สีของแอปเปลี่ยนตาม ตั้งทิ้งไว้เป็นสีที่ชอบ

  11. ใช้ Flexbox ด้วยไฟล์ CSS: สร้างไฟล์ src/pages/Home.css ถ้ายังไม่มี (ถ้ามีจากใบงาน 1.1 ให้ลบเนื้อหาเก่าทั้งหมด) แล้ววางโค้ดนี้

    ไฟล์ src/pages/Home.css
    .legend {
  display: flex;
  flex-direction: row;
  justify-content: space-between;
  align-items: center;
  gap: 8px;
  margin: 8px 16px;
  color: var(--ion-color-medium);
}
    ผลที่ควรเห็น: ยังไม่เห็นอะไรเปลี่ยน เพราะยังไม่ได้เรียกใช้ไฟล์ CSS นี้

  12. เรียกไฟล์ CSS เข้ามา: ที่ต้นไฟล์ Home.tsx ใต้บรรทัด import { add } from 'ionicons/icons'; ให้เพิ่มบรรทัดนี้

    ไฟล์ src/pages/Home.tsx
    import { add } from 'ionicons/icons';
import './Home.css';
    ผลที่ควรเห็น: ยังเหมือนเดิม (ไม่มีข้อผิดพลาด)

  13. ใช้คลาส legend: ในหน้าจอ หาบรรทัด <IonGrid> แล้วเพิ่มบรรทัดชุดนี้ไว้ก่อนมัน

    ไฟล์ src/pages/Home.tsx
    <div className="legend">
  <span>อาหาร</span>
  <span>เดินทาง</span>
  <span>การเรียน</span>
</div>

<IonGrid>
    ผลที่ควรเห็น: ระหว่างการ์ดยอดรวมกับการ์ดรายการ มีข้อความ อาหาร เดินทาง การเรียน เรียงเป็นแถวเดียว กระจายห่างกัน

  14. ทดลอง Flexbox: ในไฟล์ Home.css เปลี่ยน justify-content: space-between; เป็น center แล้วเป็น flex-end แล้วกลับเป็น space-between ดูผลทุกครั้ง

    ผลที่ควรเห็น: ข้อความทั้งสามชิดกลาง ชิดขวา และกระจายสองข้างตามลำดับ

  15. แยกการ์ดเป็นคอมโพเนนต์ ขั้นที่ 1: สร้างโฟลเดอร์ src/components (ถ้ายังไม่มี) แล้วสร้างไฟล์ ItemCard.tsx วางโค้ดนี้

    ไฟล์ src/components/ItemCard.tsx
    import { IonCard, IonCardContent, IonCardHeader, IonCardSubtitle, IonCardTitle } from '@ionic/react';

interface ItemCardProps {
  item: { id: number; title: string; category: string; amount: number };
  highlight?: boolean;
}

const ItemCard: React.FC<ItemCardProps> = ({ item, highlight = false }) => (
  <IonCard>
    <IonCardHeader>
      <IonCardSubtitle>{item.category}</IonCardSubtitle>
      <IonCardTitle>{item.title}</IonCardTitle>
    </IonCardHeader>
    <IonCardContent style={{ fontWeight: highlight ? 'bold' : 'normal' }}>{item.amount} บาท</IonCardContent>
  </IonCard>
);

export default ItemCard;
    ผลที่ควรเห็น: ยังเหมือนเดิมบนหน้าจอ เพราะหน้า Home ยังไม่เรียกใช้ ItemCard

  16. ขั้นที่ 2: ที่ Home.tsx เพิ่มบรรทัด import ItemCard ใต้บรรทัด import './Home.css';

    ไฟล์ src/pages/Home.tsx
    import './Home.css';
import ItemCard from '../components/ItemCard';
    ผลที่ควรเห็น: ยังเหมือนเดิม (ไม่มีข้อความแดง)

  17. ขั้นที่ 3: แทนที่การ์ดยาว 7 บรรทัดข้างในแต่ละ <IonCol> ด้วยบรรทัดเดียวที่เรียก ItemCard (เลือกข้อความตั้งแต่ <IonCard> ของการ์ดในตาราง จนถึง </IonCard> ตัวที่ปิดมัน แล้วแทนที่)

    ไฟล์ src/pages/Home.tsx
    <ItemCard item={i} highlight={i.amount >= 80} />
    ผลที่ควรเห็น: หน้าตาการ์ดเหมือนเดิม แต่การ์ดที่ราคา 80 บาทขึ้นไป (ซื้อสมุด 85 และ ค่าอินเทอร์เน็ต 100) ตัวเลขราคาเป็นตัวหนา (ถ้าทำขั้น 5 มาแล้วจะเป็นข้อมูลของคุณ)

  18. ลบคำที่ไม่ใช้แล้วออกจากบรรทัด import ด้านบนสุด: ลบคำว่า IonCardContent, ออก (คงเหลือ IonCard, IonCardHeader, ...) แล้วกด Ctrl+S

    ไฟล์ src/pages/Home.tsx
    IonCard, IonCardHeader,
    ผลที่ควรเห็น: ยังทำงานเหมือนเดิม ไม่มีข้อความแดง

  19. ทดสอบโหมดมืด: ใน Chrome กด F12 กดปุ่ม ⋮ (More) เลือก More tools แล้ว Rendering หาช่อง Emulate CSS media feature prefers-color-scheme เลือก dark

    ผลที่ควรเห็น: พื้นหลังและการ์ดของแอปเปลี่ยนเป็นสีเข้ม ตัวอักษรยังอ่านได้ชัด ถ้าตรงไหนอ่านยาก ให้จดไว้ แล้วเลือกกลับเป็น light

  20. ถ่ายภาพหน้าจอ 4 ภาพ: จอมือถือ 2 ขนาดต่างกัน โหมดมืด และจอ iPad (ดูเป็น 3 คอลัมน์)

    ผลที่ควรเห็น: ได้ภาพครบ 4 ภาพสำหรับส่ง

  21. เปรียบเทียบกับ wireframe หน้าหลักจากใบงาน 2.4: เขียนความแตกต่าง 2 ข้อ และวิธีปรับแต่ละข้อ ลงในช่องบันทึก

    ผลที่ควรเห็น: ได้ความแตกต่าง 2 ข้อพร้อมวิธีแก้

  22. ส่งงานขึ้น GitHub: เปิดเทอร์มินัลใหม่ (ปุ่ม +) พิมพ์ทีละบรรทัด

คำแนะนำ

  • ถ้าสีไม่เปลี่ยน ให้ตรวจว่าแก้ไฟล์ variables.css และบันทึกแล้ว และชื่อตัวแปรสะกดถูก (ขึ้นต้นด้วย --ion-color-primary)
  • ถ้าหน้าขาวหรือมีข้อความแดง ดูที่เทอร์มินัลที่รัน ionic serve และแท็บ Console ของ DevTools
