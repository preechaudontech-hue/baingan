ใบงานที่ 6.3 ออกแบบตารางจากแผนภาพคลาสและเชื่อมกับแอปของฉัน

เวลา 150 นาที · งานเดี่ยว · คะแนนเต็ม 10

จุดประสงค์

ออกแบบตารางจากแผนภาพคลาสของแอปตนเอง (อย่างน้อย 2 ตารางที่มีคีย์นอก) จัดการเวอร์ชันโครงสร้าง และเชื่อมกับหน้าจอให้ข้อมูลคงอยู่ถาวร

คำชี้แจง

นำแผนภาพคลาสจากใบงาน 2.3 มาออกแบบตารางของแอปตนเอง เขียนโครงสร้างและ migration ตามโค้ดตั้งต้น แล้วเชื่อมกับหน้าจอที่สร้างไว้ในหน่วยที่ 4 ให้อ่านและเขียนข้อมูลจริง

สิ่งที่ต้องเตรียม

  • ผลงานใบงาน 2.3 (แผนภาพคลาส) 4.4 และ 6.2
  • ใบความรู้ที่ 6 หัวข้อออกแบบตารางและเวอร์ชันโครงสร้าง

โค้ดตั้งต้น

คัดลอกไปวางในไฟล์ที่ระบุในโค้ด
// src/db/schema.ts
import type { Database } from 'sql.js';

// TODO: เปลี่ยนตารางให้ตรงกับแผนภาพคลาสของแอปตนเอง (อย่างน้อย 2 ตาราง มีคีย์นอก 1 ความสัมพันธ์)
const MIGRATIONS: Record<number, string> = {
  1: `
    CREATE TABLE IF NOT EXISTS categories (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      name TEXT NOT NULL UNIQUE
    );
    CREATE TABLE IF NOT EXISTS items (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      title TEXT NOT NULL,
      amount REAL NOT NULL CHECK (amount > 0),
      category_id INTEGER NOT NULL REFERENCES categories(id) ON DELETE RESTRICT,
      created_at TEXT NOT NULL DEFAULT (datetime('now'))
    );
    INSERT OR IGNORE INTO categories (name) VALUES ('อาหาร'), ('เดินทาง'), ('การเรียน');
  `,
  // TODO: เวอร์ชัน 2 เช่น ALTER TABLE items ADD COLUMN note TEXT
};

const LATEST_VERSION = 1;

// เรียกทุกครั้งหลังเปิดฐานข้อมูล
export function migrate(db: Database): void {
  // คีย์นอกต้องเปิดใหม่ทุกครั้งที่เปิดฐานข้อมูล
  db.run('PRAGMA foreign_keys = ON');

  const res = db.exec('PRAGMA user_version');
  let current = Number(res[0]?.values[0][0] ?? 0);
  while (current < LATEST_VERSION) {
    current += 1;
    db.exec(MIGRATIONS[current]);
    db.run(`PRAGMA user_version = ${current}`);
  }
}

ขั้นตอนการปฏิบัติ

  1. ดูแผนภาพคลาสของแอปตนเอง (ใบงาน 2.3) แล้วเขียนตารางแปลงคลาสเป็นตาราง: คลาส แอตทริบิวต์ ชื่อคอลัมน์ ชนิดข้อมูล คีย์หลัก คีย์นอก และเงื่อนไข (NOT NULL, UNIQUE, CHECK) อย่างน้อย 2 ตาราง

  2. สร้างไฟล์ src/db/schema.ts จากโค้ดตั้งต้น แก้ MIGRATIONS ข้อ 1 ให้ตรงกับการออกแบบของตนเอง และเรียก migrate(db) ใน getDb() ของใบงาน 6.2 หลังเปิดฐานข้อมูล (แทนคำสั่ง CREATE TABLE เดิม) จำไว้ว่าต้องเปลี่ยน STORAGE_KEY หรือลบข้อมูลเดิมก่อนเพราะฐานเดิมมีโครงสร้างต่างกัน

  3. เขียนฟังก์ชันในชั้น repository ให้ครบสำหรับหน้าจอของแอป (อย่างน้อย 5 ฟังก์ชัน) รวมอย่างน้อย 1 ฟังก์ชันที่ใช้ JOIN ดึงข้อมูลสองตารางมาแสดงรวมกัน (เช่น ชื่อหมวดกับรายการ)

  4. เชื่อมหน้าจอที่ทำในหน่วยที่ 4 (รายการ ฟอร์มเพิ่ม และหน้ารายละเอียด) เข้ากับฟังก์ชันเหล่านี้ ให้เพิ่ม แก้ ลบ และดูรายละเอียดข้อมูลจริงได้ทั้งแอป

  5. ทดสอบข้อกำหนดของฐานข้อมูล 4 กรณี และบันทึกข้อความผิดพลาดที่ได้: คีย์นอกผิด ลบหมวดที่ยังมีรายการอ้างอิง ค่าซ้ำในคอลัมน์ UNIQUE และค่าที่ไม่ผ่าน CHECK แล้วแสดงข้อความที่ผู้ใช้เข้าใจแทนข้อความดิบจากฐานข้อมูล

  6. ฝึก migration: เพิ่มเวอร์ชัน 2 ที่เพิ่มคอลัมน์ใหม่ 1 คอลัมน์ (เช่น note) แล้วเปลี่ยน LATEST_VERSION เป็น 2 ตรวจว่าข้อมูลที่บันทึกไว้ในเวอร์ชัน 1 ยังอยู่และคอลัมน์ใหม่ใช้ได้ (ทดสอบโดยสร้างข้อมูลก่อนแก้โค้ด)

  7. ทดสอบทั้งแอป: เพิ่มข้อมูล ปิดแท็บแล้วเปิดใหม่ ข้อมูลต้องอยู่ ถ่ายภาพหน้าจอทุกหน้าและเขียนสิ่งที่ยังไม่สมบูรณ์พร้อมแผนแก้ไข

คำแนะนำ

  • คีย์นอกจะบังคับได้ต่อเมื่อสั่ง PRAGMA foreign_keys = ON ซึ่งโค้ดตั้งต้นทำไว้ใน migrate แล้ว
  • การเพิ่มคอลัมน์ด้วย ALTER TABLE ... ADD COLUMN ที่เป็น NOT NULL ต้องกำหนดค่า DEFAULT ด้วย
  • อย่าแก้คำสั่งของเวอร์ชันที่ใช้แล้ว ให้เพิ่มเวอร์ชันใหม่เสมอ
  • ผลการสรุปด้วย JOIN และ GROUP BY ควรคำนวณด้วย SQL ไม่ใช่ดึงทุกแถวมาคำนวณใน JavaScript

สิ่งที่ต้องส่ง

ตารางแปลงคลาสเป็นตาราง + ไฟล์ schema.ts + ลิงก์ GitHub + ภาพหน้าจอทุกหน้าของแอป + ผลทดสอบข้อกำหนด 4 กรณีและ migration + สิ่งที่ยังไม่สมบูรณ์และแผนแก้ไข (ส่งตามช่องทางที่ครูกำหนด)
