จุดประสงค์

ออกแบบตารางจากแผนภาพคลาสของแอปตนเอง (อย่างน้อย 2 ตารางที่มีคีย์นอก) จัดการเวอร์ชันโครงสร้าง และเชื่อมกับหน้าจอให้ข้อมูลคงอยู่ถาวร

คำชี้แจง

ionic serve ต้องรันอยู่ ทุกขั้นที่ "วางโค้ดทั้งไฟล์" ให้ลบของเดิมก่อนด้วย Ctrl+A แล้ว Delete ข้อมูลที่เก็บเป็นของเบราว์เซอร์ที่ใช้ทดสอบ (ผูกกับที่อยู่ localhost:พอร์ต) ใบงานนี้ออกแบบตาราง 2 ตารางที่มีคีย์นอก ด้วยไฟล์โครงสร้างแยก (schema.ts) พร้อมเลขเวอร์ชัน แล้วทดสอบข้อกำหนดของฐานข้อมูล ใช้ไฟล์ใหม่ชื่อ store.ts เพื่อไม่ทับงานใบงาน 6.2

สิ่งที่ต้องเตรียม

  • ผลงานใบงาน 2.3 (แผนภาพคลาส) 4.4 และ 6.2
  • ใบความรู้ที่ 6 หัวข้อออกแบบตารางและเวอร์ชันโครงสร้าง

import type { Database } from 'sql.js';

// TODO: เปลี่ยนตารางให้ตรงกับแผนภาพคลาสของแอปตนเอง (อย่างน้อย 2 ตาราง มีคีย์นอก 1 ความสัมพันธ์)
const MIGRATIONS: Record<number, string> = {
  1: `
    CREATE TABLE IF NOT EXISTS categories (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      name TEXT NOT NULL UNIQUE
    );
    CREATE TABLE IF NOT EXISTS items (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      title TEXT NOT NULL,
      amount REAL NOT NULL CHECK (amount > 0),
      category_id INTEGER NOT NULL REFERENCES categories(id) ON DELETE RESTRICT,
      created_at TEXT NOT NULL DEFAULT (datetime('now'))
    );
    INSERT OR IGNORE INTO categories (name) VALUES ('อาหาร'), ('เดินทาง'), ('การเรียน');
  `,
  // TODO: เวอร์ชัน 2 เช่น ALTER TABLE items ADD COLUMN note TEXT

const LATEST_VERSION = 1;

// เรียกทุกครั้งหลังเปิดฐานข้อมูล
export function migrate(db: Database): void {
  // คีย์นอกต้องเปิดใหม่ทุกครั้งที่เปิดฐานข้อมูล
  db.run('PRAGMA foreign_keys = ON');

  const res = db.exec('PRAGMA user_version');
  let current = Number(res[0]?.values[0][0] ?? 0);
  while (current < LATEST_VERSION) {
    current += 1;
    db.exec(MIGRATIONS[current]);
    db.run(`PRAGMA user_version = ${current}`);
  }
}

    import initSqlJs, { Database } from 'sql.js';
import wasmUrl from 'sql.js/dist/sql-wasm.wasm?url';
import { Preferences } from '@capacitor/preferences';
import { migrate } from './schema';

const STORAGE_KEY = 'myapp-db-v2';
let db: Database | null = null;

export interface Category { id: number; name: string }
export interface ItemRow { id: number; title: string; amount: number; category: string }

function toBase64(bytes: Uint8Array): string {
  let s = '';
  for (let i = 0; i < bytes.length; i += 0x8000) {
    s += String.fromCharCode(...bytes.subarray(i, i + 0x8000));
  }
  return btoa(s);
}

function fromBase64(text: string): Uint8Array {
  const s = atob(text);
  const bytes = new Uint8Array(s.length);
  for (let i = 0; i < s.length; i++) bytes[i] = s.charCodeAt(i);
  return bytes;
}

export async function getDb(): Promise<Database> {
  if (db) return db;
  const SQL = await initSqlJs({ locateFile: () => wasmUrl });
  const saved = await Preferences.get({ key: STORAGE_KEY });
  db = saved.value ? new SQL.Database(fromBase64(saved.value)) : new SQL.Database();
  migrate(db); // สร้างหรืออัปเดตโครงสร้างตามเวอร์ชัน
  return db;
}

async function persist(): Promise<void> {
  const conn = await getDb();
  await Preferences.set({ key: STORAGE_KEY, value: toBase64(conn.export()) });
}

export async function listCategories(): Promise<Category[]> {
  const conn = await getDb();
  const stmt = conn.prepare('SELECT id, name FROM categories ORDER BY id');
  const rows: Category[] = [];
  while (stmt.step()) rows.push(stmt.getAsObject() as unknown as Category);
  stmt.free();
  return rows;
}

// JOIN สองตาราง: ดึงชื่อหมวดมาแสดงกับรายการ
export async function listItemsWithCategory(): Promise<ItemRow[]> {
  const conn = await getDb();
  const stmt = conn.prepare(
    'SELECT i.id, i.title, i.amount, c.name AS category FROM items i JOIN categories c ON c.id = i.category_id ORDER BY i.id DESC',
  );
  const rows: ItemRow[] = [];
  while (stmt.step()) rows.push(stmt.getAsObject() as unknown as ItemRow);
  stmt.free();
  return rows;
}

export async function addItem(title: string, amount: number, categoryId: number): Promise<void> {
  const conn = await getDb();
  conn.run('INSERT INTO items (title, amount, category_id) VALUES (?, ?, ?)', [title, amount, categoryId]);
  await persist();
}

export async function addCategory(name: string): Promise<void> {
  const conn = await getDb();
  conn.run('INSERT INTO categories (name) VALUES (?)', [name]);
  await persist();
}

export async function deleteCategory(id: number): Promise<void> {
  const conn = await getDb();
  conn.run('DELETE FROM categories WHERE id = ?', [id]);
  await persist();
}

// ใช้ตรวจว่า migration ทำงาน: รายชื่อคอลัมน์ของตาราง items และเลขเวอร์ชัน
export async function tableColumns(): Promise<string[]> {
  const conn = await getDb();
  const res = conn.exec('PRAGMA table_info(items)');
  return res.length === 0 ? [] : res[0].values.map(v => String(v[1]));
}

export async function schemaVersion(): Promise<number> {
  const conn = await getDb();
  const res = conn.exec('PRAGMA user_version');
  return Number(res[0]?.values[0][0] ?? 0);
}

function friendly(e: unknown): string {
  const text = String(e);
  if (text.includes('FOREIGN KEY')) return 'หมวดไม่ถูกต้อง หรือยังมีรายการที่ใช้หมวดนี้อยู่';
  if (text.includes('UNIQUE')) return 'มีชื่อนี้อยู่แล้ว';
  if (text.includes('CHECK')) return 'จำนวนเงินต้องมากกว่า 0';
  return 'เกิดข้อผิดพลาด: ' + text;
}

คำแนะนำ

  • คีย์นอกจะบังคับได้ต่อเมื่อสั่ง PRAGMA foreign_keys = ON ซึ่งโค้ดตั้งต้นทำไว้ในฟังก์ชัน migrate แล้ว
  • ALTER TABLE ... ADD COLUMN ที่เป็น NOT NULL ต้องกำหนดค่า DEFAULT ด้วย มิฉะนั้นฐานข้อมูลจะปฏิเสธ
  • ผลสรุปด้วย JOIN และ GROUP BY ควรคำนวณด้วย SQL ไม่ใช่ดึงทุกแถวมาคำนวณใน JavaScript
