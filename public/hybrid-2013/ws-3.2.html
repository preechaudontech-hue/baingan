โค้ดตั้งต้น

คัดลอกไปวางในไฟล์ที่ระบุในโค้ด
import { useState } from 'react';
import { IonContent, IonPage } from '@ionic/react';

interface Task { id: number; title: string; done: boolean }

let nextId = 1;

const Home: React.FC = () => {
  const [tasks, setTasks] = useState<Task[]>([]);
  const [newTitle, setNewTitle] = useState('');

  const remaining = tasks.filter(t => !t.done).length;

  function add() {
    // TODO: เพิ่มงานใหม่ (ไม่เพิ่มถ้าช่องว่าง) แล้วล้างช่องกรอก
  }

  function remove(id: number) {
    // TODO: ลบงานตาม id
  }

  function toggle(id: number) {
    setTasks(tasks.map(t => (t.id === id ? { ...t, done: !t.done } : t)));
  }

  return (
    <IonPage>
      <IonContent className="ion-padding">
        <h2>รายการงาน ({remaining} ที่ยังไม่เสร็จ)</h2>

        <input value={newTitle} onChange={e => setNewTitle(e.target.value)} placeholder="งานใหม่" />
        <button disabled={!newTitle.trim()} onClick={add}>เพิ่ม</button>

        {tasks.length === 0 ? (
          <p>ยังไม่มีงาน</p>
        ) : (
          <ul>
            {tasks.map(t => (
              <li key={t.id}>
                <input type="checkbox" checked={t.done} onChange={() => toggle(t.id)} />
                <span style={{ textDecoration: t.done ? 'line-through' : 'none' }}>{t.title}</span>
                <button onClick={() => remove(t.id)}>ลบ</button>
              </li>
            ))}
          </ul>
        )}
      </IonContent>
    </IonPage>
  );
};

export default Home;

ขั้นตอนการปฏิบัติ

  1. วางโค้ดตั้งต้นในไฟล์ Home.tsx แล้วดูผล (ปุ่มเพิ่มและลบยังไม่ทำงาน) ศึกษาฟังก์ชัน toggle ที่เขียนไว้ให้เป็นตัวอย่างการแก้ข้อมูลโดยสร้างสำเนาใหม่

  2. เติมฟังก์ชัน add ให้เพิ่มงานใหม่ด้วย setTasks([...tasks, ...]) (ไม่เพิ่มถ้าช่องว่างหรือมีแต่ช่องว่าง ใช้ nextId++ ให้ id ไม่ซ้ำ) แล้วล้างช่องกรอกด้วย setNewTitle('')

  3. เติมฟังก์ชัน remove ให้ลบงานตาม id ด้วย filter และตรวจว่าตัวนับเปลี่ยนตาม

  4. เพิ่มปุ่ม "ล้างงานที่เสร็จแล้ว" และข้อความ "เสร็จแล้ว x จาก y" ที่คำนวณจาก tasks โดยไม่สร้าง state เพิ่ม

  5. เพิ่มตัวเลือกกรอง ทั้งหมด / ยังไม่เสร็จ / เสร็จแล้ว ด้วย state ชนิด union ('all' | 'todo' | 'done') และกรองรายการก่อน map

  6. ใช้ console.log ในฟังก์ชัน add ดูข้อมูลใน Console ของ DevTools ถ่ายภาพ

  7. ทดสอบกรณีขอบเขต: เพิ่มชื่อซ้ำ ชื่อยาวมาก ลบทุกรายการ แล้วบันทึกพฤติกรรมที่พบ 3 ข้อ

คำแนะนำ

  • ห้ามใช้ tasks.push(...) ให้ใช้ setTasks([...tasks, item])
  • ลบโดยกรองรายการใหม่: setTasks(tasks.filter(t => t.id !== id))
  • ถ้าหน้าจอขาว ให้เปิด Console ของ DevTools ดูข้อความผิดพลาดสีแดง
