โค้ดตั้งต้น

คัดลอกไปวางในไฟล์ที่ระบุในโค้ด
import { useState } from 'react';
import {
  IonButton, IonContent, IonHeader, IonItem, IonLabel, IonList, IonPage, IonSearchbar, IonSpinner, IonTitle,
  IonToolbar,
} from '@ionic/react';
import { getJson } from '../api/http';

interface Place { name: string; latitude: number; longitude: number; country?: string }
interface GeoResponse { results?: Place[] } // ไม่พบเมือง = ไม่มีฟิลด์ results

interface Forecast {
  current: { time: string; temperature_2m: number; relative_humidity_2m: number; weather_code: number; wind_speed_10m: number };
  daily: { time: string[]; temperature_2m_max: number[]; temperature_2m_min: number[]; precipitation_probability_max: number[] };
}

// TODO: เติมข้อความของรหัสอื่นให้ครบตามตารางรหัส WMO ในเอกสาร Open-Meteo
function describe(code: number): string {
  if (code === 0) return 'ท้องฟ้าโปร่ง';
  if (code >= 1 && code <= 3) return 'มีเมฆ';
  if (code === 45 || code === 48) return 'หมอก';
  if (code >= 51 && code <= 57) return 'ฝนปรอย';
  if (code >= 61 && code <= 67) return 'ฝนตก';
  if (code >= 80 && code <= 82) return 'ฝนตกเป็นช่วง ๆ';
  if (code >= 95) return 'พายุฝนฟ้าคะนอง';
  return `รหัสสภาพอากาศ ${code}`;
}

const Weather: React.FC = () => {
  const [query, setQuery] = useState('อุดรธานี');
  const [place, setPlace] = useState<Place | null>(null);
  const [weather, setWeather] = useState<Forecast | null>(null);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState('');

  async function search() {
    const name = query.trim();
    if (!name) {
      setError('กรุณาพิมพ์ชื่อเมือง');
      return;
    }
    setLoading(true);
    setError('');
    try {
      const geoUrl = `https://geocoding-api.open-meteo.com/v1/search?name=${encodeURIComponent(name)}&count=1&language=th`;
      const geo = await getJson<GeoResponse>(geoUrl);
      const found = geo.results?.[0];
      if (!found) {
        setPlace(null);
        setWeather(null);
        setError('ไม่พบเมืองนี้ ลองพิมพ์ชื่ออื่น');
        return;
      }
      const url =
        `https://api.open-meteo.com/v1/forecast?latitude=${found.latitude}&longitude=${found.longitude}` +
        '&current=temperature_2m,relative_humidity_2m,weather_code,wind_speed_10m' +
        '&daily=temperature_2m_max,temperature_2m_min,precipitation_probability_max&timezone=auto&forecast_days=3';
      setWeather(await getJson<Forecast>(url));
      setPlace(found);
    } catch (e) {
      setError(e instanceof Error ? e.message : 'เกิดข้อผิดพลาด');
    } finally {
      setLoading(false);
    }
  }

  return (
    <IonPage>
      <IonHeader>
        <IonToolbar color="primary">
          <IonTitle>สภาพอากาศ</IonTitle>
        </IonToolbar>
      </IonHeader>
      <IonContent className="ion-padding">
        <IonSearchbar
          value={query}
          placeholder="ชื่อเมือง"
          onIonInput={e => setQuery(String(e.detail.value ?? ''))}
        />
        <IonButton expand="block" onClick={search} disabled={loading}>ดูสภาพอากาศ</IonButton>

        {loading && <p className="ion-text-center"><IonSpinner /> กำลังโหลด...</p>}
        {error && <p>{error}</p>}

        {place && weather && (
          <IonList>
            <IonItem>
              <IonLabel>
                <h2>{place.name}{place.country ? `, ${place.country}` : ''}</h2>
                <p>{describe(weather.current.weather_code)}</p>
              </IonLabel>
            </IonItem>
            <IonItem>
              <IonLabel>
                <h2>{weather.current.temperature_2m} °C</h2>
                <p>ความชื้น {weather.current.relative_humidity_2m}% · ลม {weather.current.wind_speed_10m} กม./ชม.</p>
              </IonLabel>
            </IonItem>
            {/* TODO: แสดงพยากรณ์ 3 วัน (weather.daily) วันที่ อุณหภูมิสูงสุด/ต่ำสุด และโอกาสฝน */}
          </IonList>
        )}
      </IonContent>
    </IonPage>
  );
};

export default Weather;

ขั้นตอนการปฏิบัติ

  1. สร้างไฟล์ src/api/http.ts จากใบความรู้ (โค้ดทั้งไฟล์) แล้วสร้างหน้าสภาพอากาศจากโค้ดตั้งต้น ทดสอบค้นหา "อุดรธานี" ตรวจว่าได้อุณหภูมิ

  2. เปิดแท็บ Network ใน DevTools แล้วค้นหาอีกครั้ง ดูคำขอทั้งสองที่ส่งไป (geocoding และ forecast) จดว่ามีเมธอด รหัสสถานะ และขนาดข้อมูลเท่าใด และดูตัวอย่างคำตอบ JSON ของ forecast ถ่ายภาพ

  3. เติมส่วน TODO แสดงพยากรณ์ 3 วัน (วันที่ สูงสุด/ต่ำสุด โอกาสฝน) ด้วย map และ key

  4. เติมข้อความของ describe ให้ครบอย่างน้อยกลุ่มหิมะ (71 - 77) และกลุ่มฝนตกหนัก ตามตารางรหัส WMO ในเอกสาร

  5. ทดสอบกรณีผิดปกติ 4 กรณี: ช่องว่าง ชื่อเมืองที่ไม่มีอยู่จริง (เช่น พิมพ์ว่า asdfgh) ปิดอินเทอร์เน็ตแล้วกดค้นหา (ใช้ Offline ในแท็บ Network) และเมืองที่ชื่อซ้ำกับหลายประเทศ (เช่น Udon) บันทึกข้อความที่แอปแสดง และอธิบายว่ากรณีเมืองที่ไม่มีอยู่จริงโค้ดส่วนใดป้องกันไม่ให้แอปพัง

  6. แก้ให้ผู้ใช้เลือกเมืองได้เมื่อมีหลายผลลัพธ์: ใช้ count=5 แสดงรายการเมือง (ชื่อ ประเทศ) ให้แตะเลือกแล้วจึงโหลดสภาพอากาศ

  7. เปิดเอกสาร Open-Meteo อ่านเงื่อนไขการใช้งาน แล้วเขียน 2 บรรทัดว่าแอปของเราต้องทำอะไรเพื่อใช้ API นี้อย่างถูกต้อง และเพิ่มข้อความอ้างอิงแหล่งข้อมูลไว้ท้ายหน้า

คำแนะนำ

  • ถ้าขึ้นข้อความ blocked by CORS policy ให้ตรวจ URL ว่าพิมพ์ถูกต้อง (Open-Meteo อนุญาต CORS แล้ว)
  • ชื่อเมืองภาษาไทยต้องผ่าน encodeURIComponent ซึ่งโค้ดตั้งต้นทำให้แล้ว
  • ค่า timezone=auto ทำให้เวลาของพยากรณ์ตรงกับเขตเวลาของพิกัดนั้น
  • ถ้า TypeScript ฟ้องว่า results อาจเป็น undefined แปลว่ากำลังทำถูกทาง ให้ตรวจก่อนใช้เสมอ

สิ่งที่ต้องส่ง

ภาพหน้าจอ 4 ภาพ (มีข้อมูล กำลังโหลด ไม่พบเมือง และออฟไลน์) + ภาพแท็บ Network พร้อมข้อมูลคำขอ + คำอธิบายข้อ 5 และ 7 + ลิงก์ GitHub (ส่งตามช่องทางที่ครูกำหนด)
