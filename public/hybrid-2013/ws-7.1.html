  status?: number;
  constructor(message: string, status?: number) {
    super(message);
    this.name = 'ApiError';
    this.status = status;
  }
}

const TIMEOUT_MS = 10000;

async function request<T>(url: string, init: RequestInit = {}, signal?: AbortSignal): Promise<T> {
  const ctrl = new AbortController();
  const timer = setTimeout(() => ctrl.abort(), TIMEOUT_MS);
  signal?.addEventListener('abort', () => ctrl.abort());

  let res: Response;
  try {
    res = await fetch(url, { ...init, signal: ctrl.signal });
  } catch (e) {
    if (signal?.aborted) throw e; // ผู้เรียกยกเลิกเอง (เช่น ออกจากหน้า) ส่งต่อให้ผู้เรียกจัดการ
    throw new ApiError(ctrl.signal.aborted ? 'เซิร์ฟเวอร์ตอบช้าเกินไป' : 'เชื่อมต่อเครือข่ายไม่ได้');
  } finally {
    clearTimeout(timer);
  }

  // fetch ไม่โยนข้อผิดพลาดเมื่อได้ 404 หรือ 500 ต้องตรวจ res.ok เอง
  if (!res.ok) throw new ApiError(`เซิร์ฟเวอร์ตอบกลับผิดพลาด (${res.status})`, res.status);
  if (res.status === 204) return undefined as T;

  try {
    return (await res.json()) as T;
  } catch {
    throw new ApiError('ข้อมูลที่ได้ไม่อยู่ในรูปแบบที่คาดไว้');
  }
}

export const getJson = <T>(url: string, signal?: AbortSignal) => request<T>(url, {}, signal);

export const sendJson = <T>(method: 'POST' | 'PUT' | 'PATCH' | 'DELETE', url: string, body?: unknown) =>
  request<T>(url, {
    method,
    headers: { 'Content-Type': 'application/json' },
    body: body === undefined ? undefined : JSON.stringify(body),
  });

import {

import { getJson } from '../api/http';

interface Place { name: string; latitude: number; longitude: number; country?: string }
interface GeoResponse { results?: Place[] } // ไม่พบเมือง = ไม่มีฟิลด์ results

interface Forecast {
  current: { time: string; temperature_2m: number; relative_humidity_2m: number; weather_code: number; wind_speed_10m: number };
  daily: { time: string[]; temperature_2m_max: number[]; temperature_2m_min: number[]; precipitation_probability_max: number[] };
}

// TODO: เติมข้อความของรหัสอื่นให้ครบตามตารางรหัส WMO ในเอกสาร Open-Meteo
function describe(code: number): string {
  if (code === 0) return 'ท้องฟ้าโปร่ง';
  if (code >= 1 && code <= 3) return 'มีเมฆ';
  if (code === 45 || code === 48) return 'หมอก';
  if (code >= 51 && code <= 57) return 'ฝนปรอย';
  if (code >= 61 && code <= 67) return 'ฝนตก';
  if (code >= 80 && code <= 82) return 'ฝนตกเป็นช่วง ๆ';
  if (code >= 95) return 'พายุฝนฟ้าคะนอง';
  return `รหัสสภาพอากาศ ${code}`;
}

const Weather: React.FC = () => {
  const [query, setQuery] = useState('อุดรธานี');
  const [place, setPlace] = useState<Place | null>(null);
  const [weather, setWeather] = useState<Forecast | null>(null);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState('');

  async function search() {
    const name = query.trim();
    if (!name) {
      setError('กรุณาพิมพ์ชื่อเมือง');
      return;
    }
    setLoading(true);
    setError('');
    try {
      const geoUrl = `https://geocoding-api.open-meteo.com/v1/search?name=${encodeURIComponent(name)}&count=1&language=th`;
      const geo = await getJson<GeoResponse>(geoUrl);
      const found = geo.results?.[0];
      if (!found) {
        setPlace(null);
        setWeather(null);
        setError('ไม่พบเมืองนี้ ลองพิมพ์ชื่ออื่น');
        return;
      }
      const url =
        `https://api.open-meteo.com/v1/forecast?latitude=${found.latitude}&longitude=${found.longitude}` +
        '&current=temperature_2m,relative_humidity_2m,weather_code,wind_speed_10m' +
        '&daily=temperature_2m_max,temperature_2m_min,precipitation_probability_max&timezone=auto&forecast_days=3';
      setWeather(await getJson<Forecast>(url));
      setPlace(found);
    } catch (e) {
      setError(e instanceof Error ? e.message : 'เกิดข้อผิดพลาด');
    } finally {
      setLoading(false);
    }
  }

  return (
    <IonPage>
      <IonHeader>
        <IonToolbar color="primary">
          <IonTitle>สภาพอากาศ</IonTitle>
        </IonToolbar>
      </IonHeader>
      <IonContent className="ion-padding">
        <IonSearchbar
          value={query}
          placeholder="ชื่อเมือง"
          onIonInput={e => setQuery(String(e.detail.value ?? ''))}
        />
        <IonButton expand="block" onClick={search} disabled={loading}>ดูสภาพอากาศ</IonButton>

        {loading && <p className="ion-text-center"><IonSpinner /> กำลังโหลด...</p>}
        {error && <p>{error}</p>}

        {place && weather && (
          <IonList>
            <IonItem>
              <IonLabel>
                <h2>{place.name}{place.country ? `, ${place.country}` : ''}</h2>
                <p>{describe(weather.current.weather_code)}</p>
              </IonLabel>
            </IonItem>
            <IonItem>
              <IonLabel>
                <h2>{weather.current.temperature_2m} °C</h2>
                <p>ความชื้น {weather.current.relative_humidity_2m}% · ลม {weather.current.wind_speed_10m} กม./ชม.</p>
              </IonLabel>
            </IonItem>
            {/* TODO: แสดงพยากรณ์ 3 วัน (weather.daily) วันที่ อุณหภูมิสูงสุด/ต่ำสุด และโอกาสฝน */}
          </IonList>
        )}
      </IonContent>
    </IonPage>
  );
};

export default Weather;
