ใบงานที่ 4.2 ฟอร์มรับข้อมูลและการตรวจสอบความถูกต้อง

เวลา 120 นาที · งานเดี่ยว · คะแนนเต็ม 10

จุดประสงค์

สร้างฟอร์มเพิ่มรายการด้วยช่องกรอกของ Ionic ตรวจสอบข้อมูลอย่างน้อย 4 เงื่อนไข และแจ้งผลด้วยข้อความที่เข้าใจได้

คำชี้แจง

สร้างหน้าฟอร์มเพิ่มรายการของแอปตนเอง (จาก use case ที่ออกแบบในใบงาน 2.2) ใช้โค้ดตั้งต้นแล้วปรับให้ตรงกับข้อมูลของแอป ตอนนี้ยังไม่ต้องเก็บข้อมูลถาวร (ทำในหน่วยที่ 6)

สิ่งที่ต้องเตรียม

  • ผลงานใบงาน 2.3 (ข้อมูลของแอป) และ 4.1
  • ใบความรู้ที่ 4 หัวข้อฟอร์ม

โค้ดตั้งต้น

คัดลอกไปวางในไฟล์ที่ระบุในโค้ด
import { useState } from 'react';
import {
  IonButton, IonContent, IonHeader, IonInput, IonItem, IonPage, IonSelect, IonSelectOption,
  IonTextarea, IonTitle, IonToggle, IonToolbar, useIonToast,
} from '@ionic/react';

const CATEGORIES = ['อาหาร', 'เดินทาง', 'การเรียน', 'อื่น ๆ'];

const AddItem: React.FC = () => {
  const [present] = useIonToast();
  const [title, setTitle] = useState('');
  const [amount, setAmount] = useState('');
  const [category, setCategory] = useState('');
  const [note, setNote] = useState('');
  const [monthly, setMonthly] = useState(false);
  const [touched, setTouched] = useState(false);

  // ตรวจความถูกต้อง: คืนข้อความผิดพลาด หรือข้อความว่างถ้าไม่ผิด
  const titleError = title.trim() === '' ? 'กรุณากรอกชื่อรายการ' : '';
  const amountError = Number(amount) > 0 ? '' : 'จำนวนเงินต้องมากกว่า 0';
  const categoryError = category === '' ? 'กรุณาเลือกหมวด' : '';
  const hasError = titleError !== '' || amountError !== '' || categoryError !== '';

  function submit() {
    setTouched(true);
    if (hasError) {
      present({ message: 'กรอกข้อมูลไม่ครบ', duration: 1500, color: 'danger' });
      return;
    }
    // TODO: เก็บข้อมูล (หน่วยที่ 6) ตอนนี้แสดงข้อความและล้างฟอร์ม
    present({ message: `บันทึก ${title} ${amount} บาท`, duration: 1500, color: 'success' });
    setTitle('');
    setAmount('');
    setCategory('');
    setNote('');
    setMonthly(false);
    setTouched(false);
  }

  return (
    <IonPage>
      <IonHeader>
        <IonToolbar color="primary">
          <IonTitle>เพิ่มรายการ</IonTitle>
        </IonToolbar>
      </IonHeader>
      <IonContent className="ion-padding">
        <IonInput
          label="ชื่อรายการ"
          labelPlacement="floating"
          fill="outline"
          value={title}
          onIonInput={e => setTitle(String(e.detail.value ?? ''))}
          errorText={titleError}
          className={touched && titleError ? 'ion-invalid ion-touched' : ''}
        />

        <IonInput
          label="จำนวนเงิน (บาท)"
          labelPlacement="floating"
          fill="outline"
          type="number"
          inputmode="decimal"
          value={amount}
          onIonInput={e => setAmount(String(e.detail.value ?? ''))}
          errorText={amountError}
          className={touched && amountError ? 'ion-invalid ion-touched' : ''}
        />

        <IonSelect
          label="หมวด"
          labelPlacement="floating"
          fill="outline"
          value={category}
          onIonChange={e => setCategory(String(e.detail.value ?? ''))}
        >
          {CATEGORIES.map(c => (
            <IonSelectOption key={c} value={c}>{c}</IonSelectOption>
          ))}
        </IonSelect>
        {touched && categoryError && <p style={{ color: 'var(--ion-color-danger)' }}>{categoryError}</p>}

        <IonTextarea
          label="หมายเหตุ"
          labelPlacement="floating"
          fill="outline"
          value={note}
          onIonInput={e => setNote(String(e.detail.value ?? ''))}
        />

        <IonItem lines="none">
          <IonToggle checked={monthly} onIonChange={e => setMonthly(e.detail.checked)}>
            เกิดซ้ำทุกเดือน
          </IonToggle>
        </IonItem>

        <IonButton expand="block" onClick={submit}>บันทึก</IonButton>
      </IonContent>
    </IonPage>
  );
};

export default AddItem;

ขั้นตอนการปฏิบัติ

  1. สร้างไฟล์ src/pages/AddItem.tsx จากโค้ดตั้งต้น และชั่วคราวให้ App.tsx แสดงหน้านี้ที่ path /add (เพิ่ม Route) หรือวางแทนหน้า Home เพื่อทดสอบก่อน

  2. ปรับช่องกรอกให้ตรงกับข้อมูลของแอปตนเอง (ใบงาน 2.3) อย่างน้อย 4 ช่อง ใช้ชนิดช่องให้เหมาะ (ตัวเลข ข้อความ ตัวเลือก สวิตช์)

  3. เพิ่มเงื่อนไขตรวจสอบอย่างน้อย 4 ข้อ (เช่น ห้ามว่าง ตัวเลขมากกว่า 0 ชื่อยาวไม่เกิน 40 ตัวอักษร ห้ามมีแต่ช่องว่าง) และเขียนข้อความผิดพลาดที่บอกวิธีแก้

  4. ทดสอบ: กดบันทึกโดยไม่กรอกอะไร กรอกบางช่อง กรอกครบ และกรอกค่าขอบเขต (ค่า 0 ค่าติดลบ ชื่อยาวมาก) ถ่ายภาพหน้าจออย่างน้อย 2 สถานการณ์

  5. ทำตารางทดสอบ 6 กรณี: ข้อมูลที่กรอก ผลที่คาดหวัง ผลที่ได้ และสถานะผ่านหรือไม่ผ่าน

  6. ทดสอบการเข้าถึงได้เบื้องต้น: ใช้แป้นพิมพ์กด Tab ไล่ดูทุกช่อง ตรวจว่ามีป้ายชื่อครบ และสีข้อความผิดพลาดอ่านชัดทั้งโหมดสว่างและมืด

คำแนะนำ

  • ข้อความผิดพลาดจะแสดงเมื่อช่องมีคลาส ion-invalid และ ion-touched ซึ่งโค้ดตั้งต้นใส่ให้เมื่อกดบันทึกแล้ว
  • ค่าจากช่องกรอกเป็นข้อความเสมอ ต้องแปลงด้วย Number() ก่อนเปรียบเทียบตัวเลข
  • ถ้าชื่อ prop หรือเหตุการณ์ใช้ไม่ได้หลังอัปเดตแพ็กเกจ ให้เปิดเอกสารของคอมโพเนนต์นั้นในเว็บ Ionic แล้วเทียบ
