จุดประสงค์

สร้างฟอร์มเพิ่มรายการด้วยช่องกรอกของ Ionic ตรวจสอบข้อมูลอย่างน้อย 4 เงื่อนไข และแจ้งผลด้วยข้อความที่เข้าใจได้

คำชี้แจง

ใบงานนี้ต้องรัน ionic serve ค้างไว้ และเปิดโปรเจกต์ใน VS Code ทุกขั้นที่ "วางโค้ดทั้งไฟล์" ให้คลิกในไฟล์ กด Ctrl+A แล้ว Delete ก่อนวาง ฟอร์มนี้สร้างเป็นหน้าใหม่ แล้วเพิ่มเส้นทาง /add ใน App.tsx เพื่อเปิดดู โดยไม่ทับหน้า Home จากใบงาน 4.1

สิ่งที่ต้องเตรียม

  • ผลงานใบงาน 2.3 (ข้อมูลของแอป) และ 4.1
  • ใบความรู้ที่ 4 หัวข้อฟอร์ม

ขั้นตอนการปฏิบัติ

  1. สร้างไฟล์หน้าใหม่ src/pages/AddItem.tsx (คลิกขวาที่โฟลเดอร์ pages เลือก New File) วางโค้ดขั้นที่ 1 นี้ กด Ctrl+S ขั้นนี้มีแค่ช่องชื่อรายการกับปุ่มบันทึก

    ไฟล์ src/pages/AddItem.tsx
    import { useState } from 'react';
import { IonButton, IonContent, IonHeader, IonInput, IonPage, IonTitle, IonToolbar, useIonToast } from '@ionic/react';

const AddItem: React.FC = () => {
  const [present] = useIonToast();
  const [title, setTitle] = useState('');

  function submit() {
    present({ message: `บันทึก ${title}`, duration: 1500, color: 'success' });
    setTitle('');
  }

  return (
    <IonPage>
      <IonHeader>
        <IonToolbar color="primary">
          <IonTitle>เพิ่มรายการ</IonTitle>
        </IonToolbar>
      </IonHeader>
      <IonContent className="ion-padding">
        <IonInput
          label="ชื่อรายการ"
          labelPlacement="floating"
          fill="outline"
          value={title}
          onIonInput={e => setTitle(String(e.detail.value ?? ''))}
        />
        <IonButton expand="block" onClick={submit}>บันทึก</IonButton>
      </IonContent>
    </IonPage>
  );
};

export default AddItem;
    ผลที่ควรเห็น: ยังไม่เห็นผลบนหน้าเว็บ เพราะยังไม่มีเส้นทางไปหน้านี้

  2. เพิ่มเส้นทาง: เปิดไฟล์ src/App.tsx แก้ 2 จุด จุดแรก ใต้บรรทัด import Home from './pages/Home'; เพิ่ม import AddItem from './pages/AddItem'; จุดที่สอง ใต้บรรทัด <Route path="/home" element={<Home />} /> เพิ่ม <Route path="/add" element={<AddItem />} /> แล้วกด Ctrl+S

  3. ในเบราว์เซอร์ เปลี่ยนที่อยู่ท้ายเป็น /add (เช่น http://localhost:ตัวเลข/add) แล้วกด Enter

    ผลที่ควรเห็น: เห็นหน้า เพิ่มรายการ มีช่องชื่อรายการและปุ่มบันทึกสีน้ำเงินเต็มความกว้าง

  4. พิมพ์คำว่า ข้าวมันไก่ ในช่องชื่อรายการ แล้วกดปุ่มบันทึก

    ผลที่ควรเห็น: มีข้อความสีเขียวด้านล่างจอ บันทึก ข้าวมันไก่ แล้วหายไปเองใน 1.5 วินาที และช่องถูกล้างว่าง

  5. ขั้นที่ 2: ลบทั้งไฟล์ AddItem.tsx แล้ววางโค้ดนี้ ขั้นนี้เพิ่มช่องจำนวนเงินและการตรวจข้อมูล กด Ctrl+S

    ไฟล์ src/pages/AddItem.tsx
    import { useState } from 'react';
import { IonButton, IonContent, IonHeader, IonInput, IonPage, IonTitle, IonToolbar, useIonToast } from '@ionic/react';

const AddItem: React.FC = () => {
  const [present] = useIonToast();
  const [title, setTitle] = useState('');
  const [amount, setAmount] = useState('');
  const [touched, setTouched] = useState(false);

  // คำนวณข้อความผิดพลาดจากค่าใน state ตรง ๆ (ข้อความว่าง = ไม่ผิด)
  const titleError = title.trim() === '' ? 'กรุณากรอกชื่อรายการ' : '';
  const amountError = Number(amount) > 0 ? '' : 'จำนวนเงินต้องมากกว่า 0';

  function submit() {
    setTouched(true);
    if (titleError !== '' || amountError !== '') {
      present({ message: 'กรอกข้อมูลไม่ครบ', duration: 1500, color: 'danger' });
      return;
    }
    present({ message: `บันทึก ${title} ${amount} บาท`, duration: 1500, color: 'success' });
    setTitle('');
    setAmount('');
    setTouched(false);
  }

  return (
    <IonPage>
      <IonHeader>
        <IonToolbar color="primary">
          <IonTitle>เพิ่มรายการ</IonTitle>
        </IonToolbar>
      </IonHeader>
      <IonContent className="ion-padding">
        <IonInput
          label="ชื่อรายการ"
          labelPlacement="floating"
          fill="outline"
          value={title}
          onIonInput={e => setTitle(String(e.detail.value ?? ''))}
          errorText={titleError}
          className={touched && titleError ? 'ion-invalid ion-touched' : ''}
        />

        <IonInput
          label="จำนวนเงิน (บาท)"
          labelPlacement="floating"
          fill="outline"
          type="number"
          inputmode="decimal"
          value={amount}
          onIonInput={e => setAmount(String(e.detail.value ?? ''))}
          errorText={amountError}
          className={touched && amountError ? 'ion-invalid ion-touched' : ''}
        />

        <IonButton expand="block" onClick={submit}>บันทึก</IonButton>
      </IonContent>
    </IonPage>
  );
};

export default AddItem;
    ผลที่ควรเห็น: มีช่อง จำนวนเงิน (บาท) เพิ่มอีกช่อง

  6. กดปุ่มบันทึกทันที โดยไม่กรอกอะไรเลย

    ผลที่ควรเห็น: ช่องทั้งสองมีกรอบสีแดง มีข้อความสีแดงใต้ช่อง กรุณากรอกชื่อรายการ และ จำนวนเงินต้องมากกว่า 0 และมีข้อความสีแดงด้านล่างจอ กรอกข้อมูลไม่ครบ

  7. ทดสอบ: กรอกชื่อ ข้าวมันไก่ แต่ใส่จำนวนเงินเป็น 0 แล้วกดบันทึก

    ผลที่ควรเห็น: ช่องชื่อกลับเป็นปกติ ช่องจำนวนเงินยังเป็นสีแดงพร้อมข้อความ จำนวนเงินต้องมากกว่า 0

  8. ทดสอบ: ใส่จำนวนเงินเป็น -5 แล้วเป็น abc (ลองพิมพ์ตัวอักษร ช่องรับได้เฉพาะตัวเลข) แล้วเป็น 45

    ผลที่ควรเห็น: ค่า -5 แสดงข้อผิดพลาด ตัวอักษรพิมพ์ไม่เข้า และ 45 กดบันทึกผ่าน ขึ้นข้อความสีเขียว บันทึก ข้าวมันไก่ 45 บาท และช่องว่างทั้งสอง

  9. อ่านโค้ดการตรวจข้อมูล: หา 3 บรรทัด const titleError, const amountError แล้วเขียนในช่องบันทึกว่า ทำไมข้อความผิดพลาดถึงไม่ต้องเก็บใน useState แยก (ใบความรู้ที่ 4 หัวข้อฟอร์ม)

    ผลที่ควรเห็น: ตอบได้ว่าคำนวณจากค่าใน state ได้ตรง ๆ ทุกครั้งที่หน้าวาดใหม่ จึงไม่มีทางไม่ตรงกับข้อมูล

  10. ขั้นที่ 3: ลบทั้งไฟล์แล้ววางโค้ดสมบูรณ์นี้ ขั้นนี้เพิ่มตัวเลือกหมวด ช่องหมายเหตุ สวิตช์ และตรวจหมวด กด Ctrl+S

    ไฟล์ src/pages/AddItem.tsx
    import { useState } from 'react';
import {
  IonButton, IonContent, IonHeader, IonInput, IonItem, IonPage, IonSelect, IonSelectOption,
  IonTextarea, IonTitle, IonToggle, IonToolbar, useIonToast,
} from '@ionic/react';

const CATEGORIES = ['อาหาร', 'เดินทาง', 'การเรียน', 'อื่น ๆ'];

const AddItem: React.FC = () => {
  const [present] = useIonToast();
  const [title, setTitle] = useState('');
  const [amount, setAmount] = useState('');
  const [category, setCategory] = useState('');
  const [note, setNote] = useState('');
  const [monthly, setMonthly] = useState(false);
  const [touched, setTouched] = useState(false);

  // ตรวจความถูกต้อง: คืนข้อความผิดพลาด หรือข้อความว่างถ้าไม่ผิด
  const titleError = title.trim() === '' ? 'กรุณากรอกชื่อรายการ' : '';
  const amountError = Number(amount) > 0 ? '' : 'จำนวนเงินต้องมากกว่า 0';
  const categoryError = category === '' ? 'กรุณาเลือกหมวด' : '';
  const hasError = titleError !== '' || amountError !== '' || categoryError !== '';

  function submit() {
    setTouched(true);
    if (hasError) {
      present({ message: 'กรอกข้อมูลไม่ครบ', duration: 1500, color: 'danger' });
      return;
    }
    // TODO: เก็บข้อมูล (หน่วยที่ 6) ตอนนี้แสดงข้อความและล้างฟอร์ม
    present({ message: `บันทึก ${title} ${amount} บาท`, duration: 1500, color: 'success' });
    setTitle('');
    setAmount('');
    setCategory('');
    setNote('');
    setMonthly(false);
    setTouched(false);
  }

  return (
    <IonPage>
      <IonHeader>
        <IonToolbar color="primary">
          <IonTitle>เพิ่มรายการ</IonTitle>
        </IonToolbar>
      </IonHeader>
      <IonContent className="ion-padding">
        <IonInput
          label="ชื่อรายการ"
          labelPlacement="floating"
          fill="outline"
          value={title}
          onIonInput={e => setTitle(String(e.detail.value ?? ''))}
          errorText={titleError}
          className={touched && titleError ? 'ion-invalid ion-touched' : ''}
        />

        <IonInput
          label="จำนวนเงิน (บาท)"
          labelPlacement="floating"
          fill="outline"
          type="number"
          inputmode="decimal"
          value={amount}
          onIonInput={e => setAmount(String(e.detail.value ?? ''))}
          errorText={amountError}
          className={touched && amountError ? 'ion-invalid ion-touched' : ''}
        />

        <IonSelect
          label="หมวด"
          labelPlacement="floating"
          fill="outline"
          value={category}
          onIonChange={e => setCategory(String(e.detail.value ?? ''))}
        >
          {CATEGORIES.map(c => (
            <IonSelectOption key={c} value={c}>{c}</IonSelectOption>
          ))}
        </IonSelect>
        {touched && categoryError && <p style={{ color: 'var(--ion-color-danger)' }}>{categoryError}</p>}

        <IonTextarea
          label="หมายเหตุ"
          labelPlacement="floating"
          fill="outline"
          value={note}
          onIonInput={e => setNote(String(e.detail.value ?? ''))}
        />

        <IonItem lines="none">
          <IonToggle checked={monthly} onIonChange={e => setMonthly(e.detail.checked)}>
            เกิดซ้ำทุกเดือน
          </IonToggle>
        </IonItem>

        <IonButton expand="block" onClick={submit}>บันทึก</IonButton>
      </IonContent>
    </IonPage>
  );
};

export default AddItem;
