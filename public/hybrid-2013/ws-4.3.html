ใบงานที่ 4.3 แสดงรายการ ค้นหา และลบรายการ

เวลา 120 นาที · งานเดี่ยว · คะแนนเต็ม 10

จุดประสงค์

แสดงรายการข้อมูลด้วย IonList พร้อมช่องค้นหา การลบรายการ และสถานะรายการว่าง โดยไม่แก้ state ตรง ๆ

คำชี้แจง

ใบงานนี้ต้องรัน ionic serve ค้างไว้ และเปิดโปรเจกต์ใน VS Code ทุกขั้นที่ "วางโค้ดทั้งไฟล์" ให้คลิกในไฟล์ กด Ctrl+A แล้ว Delete ก่อนวาง ใบงานนี้สร้างหน้ารายการเป็นไฟล์ใหม่ชื่อ ItemList.tsx ทีละขั้น (รายการ ค้นหา ลบ) แล้วเปิดดูที่เส้นทาง /list

สิ่งที่ต้องเตรียม

  • ผลงานใบงาน 4.1 และ 4.2
  • ใบความรู้ที่ 3 หัวข้อ useState และใบความรู้ที่ 4 หัวข้อรายการ

import { IonContent, IonHeader, IonItem, IonLabel, IonList, IonNote, IonPage, IonSearchbar, IonTitle, IonToolbar } from '@ionic/react';

interface Item { id: number; title: string; category: string; amount: number }

const ITEMS: Item[] = [
  { id: 1, title: 'ข้าวกลางวัน', category: 'อาหาร', amount: 60 },
  { id: 2, title: 'ค่ารถ', category: 'เดินทาง', amount: 40 },
  { id: 3, title: 'ซื้อสมุด', category: 'การเรียน', amount: 85 },
  { id: 4, title: 'ชานมไข่มุก', category: 'อาหาร', amount: 55 },
];

const ItemList: React.FC = () => {
  const [query, setQuery] = useState('');

  const visible = ITEMS.filter(i => i.title.includes(query) || i.category.includes(query));

  return (
    <IonPage>
      <IonHeader>
        <IonToolbar color="primary">
          <IonTitle>รายการ ({visible.length})</IonTitle>
        </IonToolbar>
        <IonToolbar>
          <IonSearchbar
            value={query}
            placeholder="ค้นหาชื่อหรือหมวด"
            onIonInput={e => setQuery(String(e.detail.value ?? ''))}
          />
        </IonToolbar>
      </IonHeader>
      <IonContent>
        {visible.length === 0 ? (
          <p className="ion-padding ion-text-center">ไม่พบรายการ</p>
        ) : (
          <IonList>
            {visible.map(i => (
              <IonItem key={i.id}>
                <IonLabel>
                  <h2>{i.title}</h2>
                  <p>{i.category}</p>
                </IonLabel>
                <IonNote slot="end">{i.amount} บาท</IonNote>
              </IonItem>
            ))}
          </IonList>
        )}
      </IonContent>
    </IonPage>
  );
};

export default ItemList;

  5. พิมพ์คำว่า อาหาร ในช่องค้นหา

  7. ลบข้อความในช่องค้นหา (กด x ในช่อง)

  8. อ่านโค้ดการค้นหา: หาบรรทัด const visible = ITEMS.filter(...) แล้วเขียนในช่องบันทึกว่า ทำไมไม่เก็บ visible เป็น state (ใบความรู้ที่ 4 หัวข้อรายการ)

    ผลที่ควรเห็น: ตอบได้ว่า visible คำนวณจาก ITEMS กับ query ได้เสมอ ไม่ต้องเก็บซ้ำ ไม่เช่นนั้นลืมอัปเดตได้

  9. ขั้นที่ 3: ลบทั้งไฟล์แล้ววางโค้ดสมบูรณ์นี้ ขั้นนี้ข้อมูลเปลี่ยนเป็น state และเพิ่มการลบด้วยการปัดแถว

    ไฟล์ src/pages/ItemList.tsx
    import { useState } from 'react';
import {
  IonContent, IonHeader, IonItem, IonItemOption, IonItemOptions, IonItemSliding, IonLabel, IonList,
  IonNote, IonPage, IonSearchbar, IonTitle, IonToolbar,
} from '@ionic/react';

interface Item { id: number; title: string; category: string; amount: number }

const INITIAL: Item[] = [
  { id: 1, title: 'ข้าวกลางวัน', category: 'อาหาร', amount: 60 },
  { id: 2, title: 'ค่ารถ', category: 'เดินทาง', amount: 40 },
  { id: 3, title: 'ซื้อสมุด', category: 'การเรียน', amount: 85 },
  { id: 4, title: 'ชานมไข่มุก', category: 'อาหาร', amount: 55 },
];

const ItemList: React.FC = () => {
  const [items, setItems] = useState<Item[]>(INITIAL);
  const [query, setQuery] = useState('');

  const visible = items.filter(i => i.title.includes(query) || i.category.includes(query));

  function remove(id: number) {
    setItems(items.filter(i => i.id !== id));
  }

  return (
    <IonPage>
      <IonHeader>
        <IonToolbar color="primary">
          <IonTitle>รายการ ({visible.length})</IonTitle>
        </IonToolbar>
        <IonToolbar>
          <IonSearchbar
            value={query}
            placeholder="ค้นหาชื่อหรือหมวด"
            onIonInput={e => setQuery(String(e.detail.value ?? ''))}
          />
        </IonToolbar>
      </IonHeader>
      <IonContent>
        {visible.length === 0 ? (
          <p className="ion-padding ion-text-center">ไม่พบรายการ</p>
        ) : (
          <IonList>
            {visible.map(i => (
              <IonItemSliding key={i.id}>
                <IonItem>
                  <IonLabel>
                    <h2>{i.title}</h2>
                    <p>{i.category}</p>
                  </IonLabel>
                  <IonNote slot="end">{i.amount} บาท</IonNote>
                </IonItem>
                <IonItemOptions side="end">
                  <IonItemOption color="danger" onClick={() => remove(i.id)}>ลบ</IonItemOption>
                </IonItemOptions>
              </IonItemSliding>
            ))}
          </IonList>
        )}
      </IonContent>
    </IonPage>
  );
};

export default ItemList;
    ผลที่ควรเห็น: หน้าตาเหมือนขั้นที่ 2

    ผลที่ควรเห็น: แถวเลื่อนออก เผยปุ่มสีแดงเขียนว่า ลบ ด้านขวา
