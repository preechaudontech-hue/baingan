ใบงานที่ 6.1 เก็บค่าตั้งค่าด้วย Preferences

เวลา 90 นาที · งานเดี่ยว · คะแนนเต็ม 10

จุดประสงค์

บันทึกและอ่านค่าตั้งค่าของแอปด้วย Preferences ให้แอปจำค่าได้หลังรีเฟรชหรือปิดเปิดใหม่ พร้อมค่าเริ่มต้นและการรีเซ็ต

คำชี้แจง

สร้างหน้าตั้งค่าของแอปตนเอง (ชื่อผู้ใช้ หมวดเริ่มต้น และตัวเลือกแสดงยอดรวม) ที่บันทึกลงเครื่องและโหลดกลับเมื่อเปิดหน้า

สิ่งที่ต้องเตรียม

  • ผลงานใบงาน 4.2 และ 5.3
  • ใบความรู้ที่ 6 หัวข้อ Preferences

โค้ดตั้งต้น

คัดลอกไปวางในไฟล์ที่ระบุในโค้ด
import { useEffect, useState } from 'react';
import {
  IonButton, IonContent, IonHeader, IonInput, IonItem, IonPage, IonSelect, IonSelectOption, IonTitle,
  IonToggle, IonToolbar, useIonToast,
} from '@ionic/react';
import { Preferences } from '@capacitor/preferences';

interface Settings { name: string; defaultCategory: string; showTotal: boolean }

const KEY = 'settings';
const DEFAULTS: Settings = { name: '', defaultCategory: 'อาหาร', showTotal: true };

async function loadSettings(): Promise<Settings> {
  const { value } = await Preferences.get({ key: KEY });
  if (!value) return DEFAULTS;
  try {
    return { ...DEFAULTS, ...JSON.parse(value) };
  } catch (e) {
    return DEFAULTS; // ข้อมูลเสียหาย ใช้ค่าเริ่มต้น
  }
}

async function saveSettings(s: Settings): Promise<void> {
  await Preferences.set({ key: KEY, value: JSON.stringify(s) });
}

const SettingsPage: React.FC = () => {
  const [settings, setSettings] = useState<Settings>(DEFAULTS);
  const [ready, setReady] = useState(false);
  const [present] = useIonToast();

  useEffect(() => {
    loadSettings().then(s => {
      setSettings(s);
      setReady(true);
    });
  }, []);

  async function save() {
    await saveSettings(settings);
    present({ message: 'บันทึกแล้ว', duration: 1200, color: 'success' });
  }

  async function reset() {
    // TODO: ลบค่าที่บันทึกไว้ด้วย Preferences.remove แล้วตั้ง settings กลับเป็น DEFAULTS
  }

  if (!ready) return <p>กำลังโหลด...</p>;

  return (
    <IonPage>
      <IonHeader>
        <IonToolbar color="primary">
          <IonTitle>ตั้งค่า</IonTitle>
        </IonToolbar>
      </IonHeader>
      <IonContent className="ion-padding">
        <h2>สวัสดี {settings.name || 'ผู้ใช้'}</h2>

        <IonInput
          label="ชื่อของคุณ"
          labelPlacement="floating"
          fill="outline"
          value={settings.name}
          onIonInput={e => setSettings({ ...settings, name: String(e.detail.value ?? '') })}
        />

        <IonSelect
          label="หมวดเริ่มต้น"
          labelPlacement="floating"
          fill="outline"
          value={settings.defaultCategory}
          onIonChange={e => setSettings({ ...settings, defaultCategory: String(e.detail.value) })}
        >
          {['อาหาร', 'เดินทาง', 'การเรียน', 'อื่น ๆ'].map(c => (
            <IonSelectOption key={c} value={c}>{c}</IonSelectOption>
          ))}
        </IonSelect>

        <IonItem lines="none">
          <IonToggle
            checked={settings.showTotal}
            onIonChange={e => setSettings({ ...settings, showTotal: e.detail.checked })}
          >
            แสดงยอดรวมในหน้าหลัก
          </IonToggle>
        </IonItem>

        <IonButton expand="block" onClick={save}>บันทึก</IonButton>
        <IonButton expand="block" fill="outline" color="medium" onClick={reset}>รีเซ็ต</IonButton>
      </IonContent>
    </IonPage>
  );
};

export default SettingsPage;

ขั้นตอนการปฏิบัติ

  1. ติดตั้งปลั๊กอินด้วยคำสั่ง npm install @capacitor/preferences แล้วสร้างหน้าตั้งค่าจากโค้ดตั้งต้น

  2. ทดสอบ: เปลี่ยนค่า กดบันทึก แล้วรีเฟรชหน้า (F5) ตรวจว่าค่าที่บันทึกไว้กลับมา เปิด DevTools แล้วดูที่แท็บ Application ส่วน Local Storage ว่าค่าที่เก็บมีคีย์ชื่ออะไร และหน้าตาของค่าเป็นอย่างไร ถ่ายภาพ

  3. เติมฟังก์ชัน reset ให้ลบค่าด้วย Preferences.remove และตั้งค่าในหน้าเป็นค่าเริ่มต้น ทดสอบว่ารีเฟรชแล้วได้ค่าเริ่มต้น

  4. ทดสอบกรณีข้อมูลเสียหาย: แก้ค่าของคีย์ settings ใน Local Storage ให้เป็นข้อความที่ไม่ใช่ JSON แล้วรีเฟรช ตรวจว่าแอปไม่พังและใช้ค่าเริ่มต้น (อธิบายว่าโค้ดส่วนใดทำให้เป็นเช่นนั้น)

  5. เพิ่มการตั้งค่าอีก 1 อย่างที่เกี่ยวกับแอปของตนเอง (เช่น สกุลเงิน หรือจำนวนรายการต่อหน้า) แล้วใช้ค่านั้นในหน้าอื่นของแอปอย่างน้อย 1 หน้า (เช่น แสดงชื่อผู้ใช้หรือซ่อนยอดรวมตามตัวเลือก)

  6. ตอบคำถาม: ข้อมูลใดในแอปของตนเองควรเก็บใน Preferences และข้อมูลใดไม่ควร เพราะอะไร (อ้างอิงตารางในใบความรู้)

คำแนะนำ

  • Preferences.get คืนค่า { value: string | null } ถ้าไม่พบคีย์จะได้ null
  • การเปลี่ยนค่าใน state ไม่ได้บันทึกอัตโนมัติ ต้องเรียก save
  • บนเว็บ Preferences ใช้ Local Storage ของเบราว์เซอร์ ค่าจึงผูกกับที่อยู่เว็บ (พอร์ต) ที่ใช้ทดสอบ

สิ่งที่ต้องส่ง

ภาพหน้าจอ 3 ภาพ (หน้าตั้งค่า ค่าที่เห็นใน DevTools และกรณีข้อมูลเสียหาย) + คำอธิบายข้อ 4 และคำตอบข้อ 6 + ลิงก์ GitHub (ส่งตามช่องทางที่ครูกำหนด)

เกณฑ์การให้คะแนน
