ใบงานที่ 5.3 เซนเซอร์วัดความเร่ง: ตรวจจับการเขย่า

เวลา 90 นาที · งานเดี่ยว · คะแนนเต็ม 10

จุดประสงค์

อ่านค่าจาก Accelerometer ตรวจจับการเขย่า และยกเลิกการสมัครรับค่าเมื่อเลิกใช้ได้

คำชี้แจง

สร้างแอปที่ตรวจจับการเขย่าเครื่อง นับจำนวนครั้ง และสั่นเตือนเบา ๆ โดยทดลองหาค่าเกณฑ์ที่เหมาะสมกับเครื่องของตนเอง

สิ่งที่ต้องเตรียม

  • ใบความรู้ที่ 5 หัวข้อ เซนเซอร์และการสั่น
  • มือถือจริง

โค้ดตั้งต้น

คัดลอกไปวางใน Snack ที่ไฟล์ App.js
import { useState, useEffect } from 'react';
import { View, Text, Button, Vibration, StyleSheet } from 'react-native';
import { Accelerometer } from 'expo-sensors';

export default function App() {
  const [available, setAvailable] = useState(null);
  const [force, setForce] = useState(0);
  const [shakes, setShakes] = useState(0);
  const [threshold, setThreshold] = useState(2.5);

  // TODO 1: useEffect ตรวจ Accelerometer.isAvailableAsync() แล้ว setAvailable
  // TODO 2: useEffect สมัครรับค่า (ตั้ง interval 200 ms) คำนวณ force = sqrt(x*x + y*y + z*z) และนับ shakes เมื่อเกิน threshold พร้อมสั่น 100 ms; ต้องมี remove() ใน cleanup และใส่ [threshold] เป็น dependency

  if (available === false) {
    return <View style={styles.container}><Text>อุปกรณ์นี้ไม่มีเซนเซอร์วัดความเร่ง</Text></View>;
  }

  return (
    <View style={styles.container}>
      <Text>ความเร่งรวม: {force.toFixed(2)} g</Text>
      <Text>เกณฑ์ตรวจจับ: {threshold.toFixed(1)} g</Text>
      <Text style={styles.big}>เขย่าแล้ว {shakes} ครั้ง</Text>
      <View style={styles.row}>
        <Button title="เกณฑ์ -" onPress={() => setThreshold(t => Math.max(1.2, t - 0.2))} />
        <Button title="เกณฑ์ +" onPress={() => setThreshold(t => t + 0.2)} />
        <Button title="รีเซ็ตจำนวน" onPress={() => setShakes(0)} />
      </View>
    </View>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, justifyContent: 'center', alignItems: 'center', gap: 10 },
  big: { fontSize: 26, fontWeight: 'bold' },
  row: { flexDirection: 'row', gap: 8 },
});

ขั้นตอนการปฏิบัติ

  1. วางโค้ดตั้งต้น เขียน useEffect ตัวแรกตรวจ isAvailableAsync แล้วตั้งค่า available

  2. เขียน useEffect ตัวที่สองสมัครรับค่าและตรวจจับการเขย่าตามคำอธิบายใน TODO และใส่ฟังก์ชันล้างที่เรียก remove()

  3. วางเครื่องนิ่งบนโต๊ะ จดค่าความเร่งรวมที่แสดง (ควรอยู่ใกล้ 1 g) แล้วเขย่าเบา ๆ เขย่าปานกลาง และเขย่าแรง จดค่าสูงสุดที่เห็นแต่ละแบบลงในตาราง

  4. ปรับเกณฑ์ด้วยปุ่ม + และ - จนเขย่าปานกลางนับได้ครั้งเดียวต่อการเขย่าหนึ่งครั้ง และวางนิ่งแล้วไม่นับ จดค่าเกณฑ์ที่เหมาะสมกับเครื่องของตน

  5. ป้องกันนับซ้ำ: เพิ่มเงื่อนไขไม่ให้นับการเขย่าซ้ำภายใน 500 มิลลิวินาที (เก็บเวลาล่าสุดด้วย useRef หรือ state) แล้วทดสอบอีกครั้ง

  6. ถ่ายภาพหน้าจอขณะนับได้อย่างน้อย 3 ครั้ง Save และคัดลอกลิงก์ และเขียนว่าถ้าไม่เรียก remove() จะเกิดปัญหาอะไร

สิ่งที่ต้องส่ง

ลิงก์ Snack + ภาพหน้าจอ + ตารางค่าความเร่งแต่ละแบบการเขย่า + ค่าเกณฑ์ที่เลือกพร้อมเหตุผล + คำอธิบายผลของการไม่ remove (ส่งตามช่องทางที่ครูกำหนด)
